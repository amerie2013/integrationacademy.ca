\begin{ideabox}
A logarithm answers "what exponent?": $\log_b x=y\iff b^y=x$. The laws turn products, quotients and powers into sums, differences and multiples. In every step write the \textbf{whole expression or equation}, and name the law you use.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item Switch forms: $\log_b x=y\iff b^y=x$; the base stays the base and the exponent is the logarithm. The argument $x$ must be positive.
  \item Product $\log_b(xy)=\log_b x+\log_b y$; Quotient $\log_b\frac{x}{y}=\log_b x-\log_b y$; Power $\log_b(x^n)=n\log_b x$.
  \item Change of base: $\log_b x=\dfrac{\log x}{\log b}$. Mistakes to avoid: $\log(x+y)\ne\log x+\log y$.
\end{itemize}
\end{ideabox}

\begin{learnbox}
\lh{A logarithm is an exponent}$\log_b x$ is the power to which $b$ must be raised to give $x$ — the inverse operation of exponentiation.
\lh{The laws}Logs turn products, quotients, and powers into sums, differences, and multiples: $\log_b(MN)=\log_b M+\log_b N$ and $\log_b M^p=p\log_b M$.
\lh{Change of base}Evaluate any logarithm on a calculator with $\log_b x=\dfrac{\log x}{\log b}$.
\lh{Writing solutions}In every step write the \textbf{whole expression}, for example $\log_2(8\cdot16)=\log_28+\log_216$, and name the law you use (product, quotient, power, change of base). The argument of a logarithm must always be positive.
\end{learnbox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Evaluate}
Find $\log_2 64$.\soln \textbf{Step 1 (ask the question):} $\log_2 64$ asks: 2 to what power gives $64$?

\textbf{Step 2 (exponential form):} Write $\log_2 64=y$ as $2^y=64$.

\textbf{Step 3 (solve):} Since $64=2^6$, the equation $2^y=64$ becomes $2^y=2^6$, so $y=6$.

\textbf{Step 4 (check):} $2^6=64$ $\checkmark$; the point $(64,6)$ is on the graph of $y=\log_2x$.

\textbf{Answer:} $\log_2 64=6$.

\end{example}

\begin{example}{ex1}{Example 2 --- Convert to log form}
Write $10^4=10000$ in logarithmic form.\soln \textbf{Step 1 (identify the parts):} In $10^4=10000$ the base is $10$, the exponent is $4$ and the result is $10000$.

\textbf{Step 2 (logarithmic form):} The exponent is the logarithm and the result is the argument: $\log_{10}10000=4$, i.e.\ $\log10000=4$ (no written base means base $10$).

\textbf{Step 3 (check):} Reverse it: $\log_{10}10000=4\iff10^4=10000$ $\checkmark$.

\textbf{Answer:} $\log10000=4$.
\end{example}

\begin{example}{ex2}{Example 3 --- Evaluate}
Find $\log_5 25$.\soln \textbf{Step 1 (exponential form):} Write $\log_525=y$ as $5^y=25$.

\textbf{Step 2 (solve):} Since $25=5^2$, $5^y=25$ becomes $5^y=5^2$, so $y=2$.

\textbf{Step 3 (check):} $5^2=25$ $\checkmark$; the points $(5,1)$ and $(25,2)$ are on $y=\log_5x$.

\textbf{Answer:} $\log_525=2$.

\begin{example}{ex3}{Example 4 --- Product law}
Simplify $\log_2(8\cdot16)$.\soln \textbf{Step 1 (product law):} $\log_2(8\cdot16)=\log_28+\log_216$.

\textbf{Step 2 (evaluate each term):} $\log_28+\log_216=3+4=7$, because $2^3=8$ and $2^4=16$.

\textbf{Step 3 (check):} $8\cdot16=128=2^7$, so $\log_2128=7$ $\checkmark$.

\textbf{Answer:} $\log_2(8\cdot16)=7$.

\end{example}

\begin{example}{ex4}{Example 5 --- Power law}
Simplify $\log_3(9^3)$.\soln \textbf{Step 1 (power law):} $\log_3(9^3)=3\log_39$.

\textbf{Step 2 (evaluate):} $3\log_39=3\cdot2=6$, because $3^2=9$.

\textbf{Step 3 (check):} $9^3=(3^2)^3=3^6$, so $\log_3(3^6)=6$ $\checkmark$.

\textbf{Answer:} $\log_3(9^3)=6$.
\end{example}

\begin{example}{ex5}{Example 6 --- Evaluate}
Find $\log_7 49$.\soln \textbf{Step 1 (exponential form):} Write $\log_749=y$ as $7^y=49$.

\textbf{Step 2 (solve):} Since $49=7^2$, $7^y=7^2$ gives $y=2$.

\textbf{Step 3 (check):} $7^2=49$ $\checkmark$; the point $(49,2)$ is on $y=\log_7x$.

\textbf{Answer:} $\log_749=2$.

\begin{example}{ex6}{Example 7 --- Base 10}
Find $\log 100000$.\soln \textbf{Step 1 (base):} $\log100000$ has no written base, so it is $\log_{10}100000$.

\textbf{Step 2 (exponential form):} Write $\log_{10}100000=y$ as $10^y=100000=10^5$, so $y=5$.

\textbf{Step 3 (check):} Count the zeros: $100000$ has $5$ zeros, so $\log100000=5$ $\checkmark$.

\textbf{Answer:} $\log100000=5$.
\end{example}

\begin{example}{ex7}{Example 8 --- Quotient law}
Simplify $\log_2\dfrac{32}{4}$.\soln \textbf{Step 1 (quotient law):} $\log_2\dfrac{32}{4}=\log_232-\log_24$.

\textbf{Step 2 (evaluate):} $\log_232-\log_24=5-2=3$.

\textbf{Step 3 (check):} $\dfrac{32}{4}=8=2^3$, so $\log_28=3$ $\checkmark$.

\textbf{Answer:} $\log_2\dfrac{32}{4}=3$.

\end{example}

\begin{example}{ex8}{Example 9 --- Change of base}
Express $\log_2 10$ with base-10 logs and evaluate it.\soln \textbf{Step 1 (change of base):} $\log_210=\dfrac{\log10}{\log2}$.

\textbf{Step 2 (calculate):} $\dfrac{\log10}{\log2}=\dfrac{1}{0.3010}\approx3.32$.

\textbf{Step 3 (check):} $2^3=8<10<16=2^4$, so an exponent between $3$ and $4$ is reasonable $\checkmark$.

\textbf{Answer:} $\log_210\approx3.32$.

\end{example}
