\wstitle{4.4}{Related Rates: Setting Up}{Unit 4: Derivatives in Context}

\begin{ideabox}
\begin{itemize}[leftmargin=*,itemsep=2pt,topsep=2pt]
  \item Name the changing quantities, and write the given and wanted rates as derivatives with respect to time, with signs.
  \item Write \textbf{one equation} that relates the quantities at every instant.
  \item Differentiate both sides with respect to $t$. Every changing quantity produces its own rate.
  \item Substitute the values for the instant in question \textbf{only after} differentiating.
\end{itemize}
\end{ideabox}

\sech{Worked examples}
\begin{example}{Example 1 \textperiodcentered{} A sphere}
The radius of a sphere changes with time. Relate $\dfrac{dV}{dt}$ to $\dfrac{dr}{dt}$.\soln $V=\dfrac43\pi r^3$.\result{$\dfrac{dV}{dt}=4\pi r^2\dfrac{dr}{dt}$.}
\end{example}

\begin{example}{Example 2 \textperiodcentered{} A fixed hypotenuse}
A right triangle has legs $x$ and $y$ and a hypotenuse of fixed length 10. Relate $\dfrac{dx}{dt}$ and $\dfrac{dy}{dt}$.\soln $x^2+y^2=100$, so $2x\dfrac{dx}{dt}+2y\dfrac{dy}{dt}=0$.\result{$x\dfrac{dx}{dt}+y\dfrac{dy}{dt}=0$.}
\end{example}

\begin{example}{Example 3 \textperiodcentered{} Two changing factors}
A triangle has base $x$ and height $y$, both changing. Find $\dfrac{dA}{dt}$.\soln $A=\dfrac12xy$; use the product rule.\result{$\dfrac{dA}{dt}=\dfrac12\left(\dfrac{dx}{dt}y+x\dfrac{dy}{dt}\right)$.}
\end{example}

\sech{Practice}

\level{Core}
\begin{enumerate}[leftmargin=*,itemsep=2pt]
  \item The radius of a circle is increasing at 3 cm/s. How fast is its circumference increasing?\work{2.3cm}
  \item The edge of a cube is increasing at 0.5 cm/s. How fast is its volume increasing when the edge is 4 cm?\work{2.3cm}
  \item A point moves along $y=x^2+1$ with $\dfrac{dx}{dt}=3$. Find $\dfrac{dy}{dt}$ when $x=2$.\work{2.3cm}
  \item Write as an equation: \textquotedblleft{}the radius is decreasing at 2 centimetres per second\textquotedblright{}.\work{2.3cm}
\end{enumerate}

\level{Exam level}
\begin{enumerate}[leftmargin=*,itemsep=2pt,start=5]
  \item The area of a square is increasing at 20 cm$^2$/s. How fast is its side increasing when the side is 5 cm?\work{2.8cm}
  \item Two quantities satisfy $xy=24$. When $x=4$, $x$ is decreasing at 1 unit per second. Find $\dfrac{dy}{dt}$.\work{2.8cm}
  \item The volume of a sphere is increasing at $100\pi$ cm$^3$/s. How fast is its radius increasing when $r=5$ cm?\work{2.8cm}
  \item Quantities satisfy $z^2=x^2+y^2$ with $z>0$. When $x=8$ and $y=6$, $\dfrac{dx}{dt}=1$ and $\dfrac{dy}{dt}=3$. Find $\dfrac{dz}{dt}$.\work{2.8cm}
\end{enumerate}

\begin{frbox}{Question 9 \textperiodcentered{} Free-response practice}
The length $\ell$ of a rectangle is increasing at 4 cm/s and its width $w$ is decreasing at 3 cm/s. At a certain instant $\ell=12$ cm and $w=5$ cm.
\begin{enumerate}[label=(\alph*),leftmargin=*,itemsep=2pt]
  \item Find the rate at which the area of the rectangle is changing at that instant.\work{2.6cm}
  \item Find the rate at which the perimeter is changing.\work{2.6cm}
  \item Find the rate at which the length of the diagonal is changing at that instant.\work{2.6cm}
\end{enumerate}
\end{frbox}

\newpage
\begin{solbox}
\begin{enumerate}[leftmargin=*,itemsep=4pt]
  \item $\dfrac{dC}{dt}=2\pi\dfrac{dr}{dt}=6\pi$ cm/s.
  \item $\dfrac{dV}{dt}=3s^2\dfrac{ds}{dt}=3(16)(0.5)=24$ cm$^3$/s.
  \item $\dfrac{dy}{dt}=2x\dfrac{dx}{dt}=12$.
  \item $\dfrac{dr}{dt}=-2$.
  \item $\dfrac{dA}{dt}=2s\dfrac{ds}{dt}$: $20=10\dfrac{ds}{dt}$, so 2 cm/s.
  \item $y=6$. $\dfrac{dx}{dt}y+x\dfrac{dy}{dt}=0$: $-6+4\dfrac{dy}{dt}=0$, so $\dfrac{dy}{dt}=1.5$.
  \item $100\pi=4\pi(25)\dfrac{dr}{dt}$, so 1 cm/s.
  \item $z=10$. $10\dfrac{dz}{dt}=8(1)+6(3)=26$, so $\dfrac{dz}{dt}=2.6$.
  \item \begin{enumerate}[label=(\alph*),leftmargin=*,itemsep=3pt]
    \item $\dfrac{dA}{dt}=\dfrac{d\ell}{dt}w+\ell\dfrac{dw}{dt}=4(5)+12(-3)=-16$ cm$^2$/s.
    \item $\dfrac{dP}{dt}=2\left(\dfrac{d\ell}{dt}+\dfrac{dw}{dt}\right)=2(4-3)=2$ cm/s.
    \item $D=13$. $D\dfrac{dD}{dt}=\ell\dfrac{d\ell}{dt}+w\dfrac{dw}{dt}=48-15=33$, so $\dfrac{dD}{dt}=\dfrac{33}{13}$ cm/s.
  \end{enumerate}
\end{enumerate}
\end{solbox}
