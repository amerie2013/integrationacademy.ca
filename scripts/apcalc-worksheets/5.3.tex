\wstitle{5.3}{Increasing and Decreasing Intervals}{Unit 5: Analysing Functions}

\begin{ideabox}
\begin{itemize}[leftmargin=*,itemsep=2pt,topsep=2pt]
  \item $f$ is \textbf{increasing} on an interval where $f'(x)>0$ and \textbf{decreasing} where $f'(x)<0$.
  \item Find the critical points, then test the sign of $f'$ on each interval between them.
  \item A factor that is always positive (a square, an exponential) does not affect the sign.
  \item The justification is a statement about the sign of $f'$ on the whole interval.
\end{itemize}
\end{ideabox}

\sech{Worked examples}
\begin{example}{Example 1 \textperiodcentered{} A cubic}
Find where $f(x)=2x^3-3x^2-12x$ is increasing and decreasing.\soln $f'(x)=6x^2-6x-12=6(x-2)(x+1)$, with signs $+,-,+$.\result{Increasing on $(-\infty,-1)$ and $(2,\infty)$; decreasing on $(-1,2)$.}
\end{example}

\begin{example}{Example 2 \textperiodcentered{} A logarithm}
Find where $f(x)=x-2\ln x$, $x>0$, is increasing and decreasing.\soln $f'(x)=1-\dfrac2x=\dfrac{x-2}{x}$.\result{Decreasing on $(0,2)$; increasing on $(2,\infty)$.}
\end{example}

\begin{example}{Example 3 \textperiodcentered{} From the derivative}
A function has $f'(x)=(x-1)(x+2)^2$. Where is $f$ increasing?\soln $(x+2)^2\ge0$, so the sign of $f'$ is the sign of $x-1$.\result{For $x>1$.}
\end{example}

\sech{Practice}

\level{Core}
\begin{enumerate}[leftmargin=*,itemsep=2pt]
  \item Find where $f(x)=x^2-6x$ is decreasing.\work{2.3cm}
  \item Show that $f(x)=-x^3$ is never increasing.\work{2.3cm}
  \item Find where $f(x)=e^{x}-x$ is increasing.\work{2.3cm}
  \item A function has $f'(x)=x(x-4)$. On what intervals is $f$ increasing?\work{2.3cm}
\end{enumerate}

\level{Exam level}
\begin{enumerate}[leftmargin=*,itemsep=2pt,start=5]
  \item Find where $f(x)=3x^4-8x^3$ is decreasing.\work{2.8cm}
  \item Show that $f(x)=x+\cos x$ is never decreasing.\work{2.8cm}
  \item Find where $f(x)=x+\dfrac3x$ is decreasing.\work{2.8cm}
  \item Find where $f(x)=x^2-2\ln x$, $x>0$, is increasing.\work{2.8cm}
\end{enumerate}

\begin{frbox}{Question 9 \textperiodcentered{} Free-response practice}
Let $f(x)=(x-3)^2e^{x}$.
\begin{enumerate}[label=(\alph*),leftmargin=*,itemsep=2pt]
  \item Find $f'(x)$ in factored form.\work{2.6cm}
  \item Find the intervals on which $f$ is increasing and the interval on which $f$ is decreasing. Justify your answer.\work{2.6cm}
  \item Without computing, explain why $f(2)<f(1)$.\work{2.6cm}
\end{enumerate}
\end{frbox}

\newpage
\begin{solbox}
\begin{enumerate}[leftmargin=*,itemsep=4pt]
  \item $f'(x)=2x-6<0$ for $x<3$.
  \item $f'(x)=-3x^2\le0$ for all $x$.
  \item $f'(x)=e^{x}-1>0$ for $x>0$.
  \item $(-\infty,0)$ and $(4,\infty)$.
  \item $f'(x)=12x^3-24x^2=12x^2(x-2)<0$ for $x<2$ ($x\neq0$): decreasing on $(-\infty,2)$.
  \item $f'(x)=1-\sin x\ge0$ for all $x$.
  \item $f'(x)=1-\dfrac{3}{x^2}<0$ when $x^2<3$: on $\left(-\sqrt3,0\right)$ and $\left(0,\sqrt3\right)$.
  \item $f'(x)=2x-\dfrac2x=\dfrac{2(x^2-1)}{x}>0$ for $x>1$.
  \item \begin{enumerate}[label=(\alph*),leftmargin=*,itemsep=3pt]
    \item $2(x-3)e^{x}+(x-3)^2e^{x}=(x-3)(x-1)e^{x}$.
    \item $f'>0$ for $x<1$ and $x>3$: increasing on $(-\infty,1)$ and $(3,\infty)$. $f'<0$ for $1<x<3$: decreasing on $(1,3)$.
    \item $f$ is decreasing on $[1,3]$ because $f'(x)<0$ on $(1,3)$, and $1<2$.
  \end{enumerate}
\end{enumerate}
\end{solbox}
