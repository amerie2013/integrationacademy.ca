\wstitle{5.4}{The First Derivative Test}{Unit 5: Analysing Functions}

\begin{ideabox}
\begin{itemize}[leftmargin=*,itemsep=2pt,topsep=2pt]
  \item At a critical point $c$: if $f'$ changes from \textbf{positive to negative}, $f$ has a relative \textbf{maximum}; from \textbf{negative to positive}, a relative \textbf{minimum}.
  \item If $f'$ does not change sign at $c$, there is no extremum there.
  \item The location is $x=c$; the extreme \emph{value} is $f(c)$. Read which one the question wants.
  \item \textquotedblleft{}$f'(c)=0$\textquotedblright{} is not a justification. The sign change is.
\end{itemize}
\end{ideabox}

\sech{Worked examples}
\begin{example}{Example 1 \textperiodcentered{} A cubic}
Find the relative extrema of $f(x)=x^3-6x^2+9x$.\soln $f'(x)=3(x-1)(x-3)$, with signs $+,-,+$.\result{Relative maximum $f(1)=4$; relative minimum $f(3)=0$.}
\end{example}

\begin{example}{Example 2 \textperiodcentered{} A repeated factor}
A function has $f'(x)=(x+1)(x-2)^2(x-5)$. Classify its critical points.\soln Signs of $f'$: $+$ for $x<-1$; $-$ on $(-1,2)$ and on $(2,5)$; $+$ for $x>5$.\result{Maximum at $x=-1$, minimum at $x=5$, neither at $x=2$.}
\end{example}

\begin{example}{Example 3 \textperiodcentered{} A cusp-type critical point}
Classify the critical points of $f(x)=x-3x^{1/3}$.\soln $f'(x)=1-x^{-2/3}$ is positive for $|x|>1$ and negative for $0<|x|<1$.\result{Maximum at $x=-1$ (value 2), minimum at $x=1$ (value $-2$), neither at $x=0$.}
\end{example}

\sech{Practice}

\level{Core}
\begin{enumerate}[leftmargin=*,itemsep=2pt]
  \item Find the relative minimum value of $f(x)=x^2-4x+7$.\work{2.3cm}
  \item Find the relative maximum value of $f(x)=6x-x^2$.\work{2.3cm}
  \item A function has $f'$ changing from negative to positive at $x=4$. What does $f$ have there?\work{2.3cm}
  \item A function has $f'(x)=(x-3)^2$. Does $f$ have a relative extremum at $x=3$?\work{2.3cm}
\end{enumerate}

\level{Exam level}
\begin{enumerate}[leftmargin=*,itemsep=2pt,start=5]
  \item Find and classify the relative extremum of $f(x)=x^4-4x$.\work{2.8cm}
  \item Find and classify the relative extremum of $f(x)=xe^{-x/3}$.\work{2.8cm}
  \item Find and classify the relative extremum of $f(x)=x^2+\dfrac2x$ for $x>0$.\work{2.8cm}
  \item Find and classify the relative extrema of $f(x)=\sqrt3\sin x+\cos x$ on $(0,2\pi)$.\work{2.8cm}
\end{enumerate}

\begin{frbox}{Question 9 \textperiodcentered{} Free-response practice}
Let $f(x)=x^3e^{-x}$.
\begin{enumerate}[label=(\alph*),leftmargin=*,itemsep=2pt]
  \item Find $f'(x)$ in factored form.\work{2.6cm}
  \item Find the relative maximum value of $f$. Justify your answer.\work{2.6cm}
  \item Explain why $f$ has no relative extremum at $x=0$.\work{2.6cm}
\end{enumerate}
\end{frbox}

\newpage
\begin{solbox}
\begin{enumerate}[leftmargin=*,itemsep=4pt]
  \item $f'(x)=2x-4$ changes from negative to positive at $x=2$: minimum value $f(2)=3$.
  \item $f'(x)=6-2x$ changes from positive to negative at $x=3$: maximum value $f(3)=9$.
  \item A relative minimum.
  \item No: $f'$ is positive on both sides of 3.
  \item $f'(x)=4x^3-4$ changes from negative to positive at $x=1$: relative minimum, value $-3$.
  \item $f'(x)=e^{-x/3}\left(1-\dfrac x3\right)$ changes from positive to negative at $x=3$: relative maximum, value $\dfrac3e$.
  \item $f'(x)=2x-\dfrac{2}{x^2}$ changes from negative to positive at $x=1$: relative minimum, value 3.
  \item $f'(x)=\sqrt3\cos x-\sin x=0$ when $\tan x=\sqrt3$. Maximum at $x=\dfrac\pi3$ (value 2); minimum at $x=\dfrac{4\pi}{3}$ (value $-2$).
  \item \begin{enumerate}[label=(\alph*),leftmargin=*,itemsep=3pt]
    \item $3x^2e^{-x}-x^3e^{-x}=x^2(3-x)e^{-x}$.
    \item $f'$ changes from positive to negative at $x=3$, so the relative maximum value is $f(3)=\dfrac{27}{e^3}$.
    \item $f'(0)=0$, but $x^2\ge0$ means $f'$ is positive on both sides of 0: no sign change.
  \end{enumerate}
\end{enumerate}
\end{solbox}
