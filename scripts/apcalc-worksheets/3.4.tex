\wstitle{3.4}{Derivatives of Inverse Trigonometric Functions}{Unit 3: Composite, Implicit and Inverse Functions}

\begin{ideabox}
\begin{itemize}[leftmargin=*,itemsep=2pt,topsep=2pt]
  \item $\dfrac{d}{dx}[\arcsin x]=\dfrac{1}{\sqrt{1-x^2}}$ \quad and \quad $\dfrac{d}{dx}[\arccos x]=-\dfrac{1}{\sqrt{1-x^2}}$.
  \item $\dfrac{d}{dx}[\arctan x]=\dfrac{1}{1+x^2}$.
  \item With an inside function $u$, replace $x$ by $u$ and multiply by $u'$: \ $\dfrac{d}{dx}[\arctan u]=\dfrac{u'}{1+u^2}$.
  \item Square the \emph{whole} inside: for $\arctan(3x)$ the denominator is $1+9x^2$.
\end{itemize}
\end{ideabox}

\sech{Worked examples}
\begin{example}{Example 1 \textperiodcentered{} With the chain rule}
Find $\dfrac{d}{dx}[\arcsin(2x)]$.\soln $\dfrac{1}{\sqrt{1-(2x)^2}}\cdot2$.\result{$\dfrac{2}{\sqrt{1-4x^2}}$.}
\end{example}

\begin{example}{Example 2 \textperiodcentered{} With the product rule}
Find $\dfrac{d}{dx}[x\arccos x]$.\soln $(1)\arccos x+x\cdot\left(-\dfrac{1}{\sqrt{1-x^2}}\right)$.\result{$\arccos x-\dfrac{x}{\sqrt{1-x^2}}$.}
\end{example}

\begin{example}{Example 3 \textperiodcentered{} A tangent line}
Write an equation of the line tangent to $y=\arccos x$ at $x=\dfrac12$.\soln The point is $\left(\dfrac12,\dfrac\pi3\right)$. The slope is $-\dfrac{1}{\sqrt{1-\frac14}}=-\dfrac{2}{\sqrt3}$.\result{$y-\dfrac\pi3=-\dfrac{2}{\sqrt3}\left(x-\dfrac12\right)$.}
\end{example}

\begin{frbox}{Question 9 \textperiodcentered{} Free-response practice}
Let $f(x)=\arctan(2x)$.
\begin{enumerate}[label=(\alph*),leftmargin=*,itemsep=2pt]
  \item Find $f'(x)$.\work{2.6cm}
  \item Write an equation of the line tangent to the graph of $f$ at $x=\dfrac12$.\work{2.6cm}
  \item Find all values of $x$ at which the slope of the graph of $f$ is $\dfrac25$.\work{2.6cm}
\end{enumerate}
\end{frbox}

\begin{solbox}
\begin{enumerate}[leftmargin=*,itemsep=4pt]
  \item $\dfrac{4}{1+x^2}$.
  \item $-\dfrac{3}{\sqrt{1-9x^2}}$.
  \item $\dfrac{5}{1+25x^2}$.
  \item $\dfrac{1}{1+4}=\dfrac15$.
  \item $\dfrac{e^{x}}{\sqrt{1-e^{2x}}}$.
  \item $\dfrac{1}{1+\frac{1}{x^2}}\cdot\left(-\dfrac{1}{x^2}\right)=-\dfrac{1}{x^2+1}$.
  \item $\dfrac{2\arcsin x}{\sqrt{1-x^2}}$.
  \item $f'(x)=\dfrac{1/3}{1+x^2/9}$, so $f'(3)=\dfrac{1/3}{2}=\dfrac16$.
  \item \begin{enumerate}[label=(\alph*),leftmargin=*,itemsep=3pt]
    \item $f'(x)=\dfrac{2}{1+4x^2}$.
    \item $f\!\left(\dfrac12\right)=\arctan1=\dfrac\pi4$ and $f'\!\left(\dfrac12\right)=1$: $y-\dfrac\pi4=x-\dfrac12$.
    \item $\dfrac{2}{1+4x^2}=\dfrac25$ gives $1+4x^2=5$, so $x=\pm1$.
  \end{enumerate}
\end{enumerate}
\end{solbox}
