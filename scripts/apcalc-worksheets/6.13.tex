\wstitle{6.13}{Unit 6 Synthesis: The Accumulation Free-Response Question}{Unit 6: Integration and Accumulation of Change}

\begin{ideabox}
\begin{itemize}[leftmargin=*,itemsep=2pt,topsep=2pt]
  \item Amount at time $t$ $=$ starting amount $+\ \displaystyle\int_{0}^{t}\text{rate}\,ds$.
  \item The derivative of an accumulation function is its integrand, so its extrema are where the integrand changes sign.
  \item On a closed interval, compare the values at critical points \textbf{and} endpoints.
  \item Write the integral before evaluating it: the set-up earns a point of its own.
\end{itemize}
\end{ideabox}

\sech{Worked examples}
\begin{example}{Example 1 \textperiodcentered{} An amount from a rate}
A tank holds 20 litres at $t=0$, and water flows in at the net rate $6-2t$ litres per minute. Find the amount at $t=4$ and the greatest amount for $0\le t\le4$.\soln $A(t)=20+\displaystyle\int_{0}^{t}(6-2s)\,ds=20+6t-t^2$. $A'(t)=6-2t=0$ at $t=3$.\result{$A(4)=28$; the greatest amount is $A(3)=29$ litres.}
\end{example}

\begin{example}{Example 2 \textperiodcentered{} Differentiating an accumulation}
Find $\dfrac{d}{dx}\displaystyle\int_{1}^{x}te^{t}\,dt$.\soln Fundamental Theorem, Part 1.\result{$xe^{x}$.}
\end{example}

\begin{example}{Example 3 \textperiodcentered{} A definite substitution}
Evaluate $\displaystyle\int_{0}^{1}6x\left(x^2+1\right)^2\,dx$.\soln $u=x^2+1$ from 1 to 2: $\displaystyle\int_{1}^{2}3u^2\,du$.\result{$8-1=7$.}
\end{example}

\sech{Practice}

\level{Core}
\begin{enumerate}[leftmargin=*,itemsep=2pt]
  \item Evaluate $\displaystyle\int_{0}^{3}(2x+1)\,dx$.\work{2.3cm}
  \item Find $\displaystyle\int\sec^2(2x)\,dx$.\work{2.3cm}
  \item Find $\dfrac{d}{dx}\displaystyle\int_{0}^{x}\sqrt{t^2+9}\,dt$ and its value at $x=4$.\work{2.3cm}
  \item A differentiable function $f$ has $f(1)=3$ and $f(4)=-2$. Find $\displaystyle\int_{1}^{4}f'(x)\,dx$.\work{2.3cm}
\end{enumerate}

\level{Exam level}
\begin{enumerate}[leftmargin=*,itemsep=2pt,start=5]
  \item Water flows into a pond at the net rate $R(t)=12-3t$ litres per minute for $0\le t\le6$. Find the net change in the amount of water over the 6 minutes, and the time at which the amount is greatest.\work{2.8cm}
  \item Evaluate $\displaystyle\int_{0}^{2}x\sqrt{4-x^2}\,dx$.\work{2.8cm}
  \item Let $F(x)=\displaystyle\int_{0}^{x}(t-2)e^{t}\,dt$. At what value of $x$ does $F$ have a relative minimum? Justify your answer.\work{2.8cm}
  \item A particle has velocity $v(t)=t^2-4$ for $0\le t\le3$. Find the total distance it travels.\work{2.8cm}
\end{enumerate}

\begin{frbox}{Question 9 \textperiodcentered{} Free-response practice}
Let $G(x)=\displaystyle\int_{0}^{x}\left(3t^2-12\right)\,dt$ for $0\le x\le4$.
\begin{enumerate}[label=(\alph*),leftmargin=*,itemsep=2pt]
  \item Find $G'(x)$ and $G''(x)$.\work{2.6cm}
  \item Find the $x$-coordinate of the relative minimum of $G$, justify it, and find the value of $G$ there.\work{2.6cm}
  \item Find the absolute maximum value of $G$ on $[0,4]$. Justify your answer.\work{2.6cm}
\end{enumerate}
\end{frbox}

\newpage
\begin{solbox}
\begin{enumerate}[leftmargin=*,itemsep=4pt]
  \item $\Big[x^2+x\Big]_0^3=12$.
  \item $\dfrac12\tan(2x)+C$.
  \item $\sqrt{x^2+9}$; at $x=4$ it is $5$.
  \item $f(4)-f(1)=-5$.
  \item $\displaystyle\int_{0}^{6}(12-3t)\,dt=72-54=18$ litres. $R$ changes from positive to negative at $t=4$, so the amount is greatest then.
  \item $u=4-x^2$ from 4 to 0, $x\,dx=-\dfrac12du$: $\dfrac12\displaystyle\int_{0}^{4}\sqrt u\,du=\dfrac12\cdot\dfrac23(8)=\dfrac83$.
  \item $F'(x)=(x-2)e^{x}$ changes from negative to positive at $x=2$.
  \item $v$ changes sign at $t=2$. $\displaystyle\int_{0}^{2}(t^2-4)\,dt=-\dfrac{16}{3}$ and $\displaystyle\int_{2}^{3}(t^2-4)\,dt=\dfrac73$. Distance $\dfrac{16}{3}+\dfrac73=\dfrac{23}{3}$.
  \item \begin{enumerate}[label=(\alph*),leftmargin=*,itemsep=3pt]
    \item $G'(x)=3x^2-12$ and $G''(x)=6x$.
    \item $G'(x)=3(x-2)(x+2)$ changes from negative to positive at $x=2$. $G(2)=\Big[t^3-12t\Big]_0^2=-16$.
    \item Candidates: $G(0)=0$, $G(2)=-16$, $G(4)=64-48=16$. The absolute maximum is 16, at $x=4$.
  \end{enumerate}
\end{enumerate}
\end{solbox}
