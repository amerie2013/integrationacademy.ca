\wstitle{5.8}{Sketching a Function from Its Derivatives}{Unit 5: Analysing Functions}

\begin{ideabox}
\begin{itemize}[leftmargin=*,itemsep=2pt,topsep=2pt]
  \item Collect, in order: domain and intercepts; asymptotes; where $f'$ is positive or negative; where $f''$ is positive or negative.
  \item The signs of $f'$ and $f''$ together give one of four shapes on each interval: rising or falling, bending up or down.
  \item A horizontal asymptote comes from a limit as $x\to\pm\infty$; a vertical asymptote from an infinite one-sided limit.
  \item Mark relative extrema and points of inflection as points, with both coordinates.
\end{itemize}
\end{ideabox}

\sech{Worked examples}
\begin{example}{Example 1 \textperiodcentered{} A rational function}
Describe the graph of $f(x)=\dfrac{x}{x-2}$.\soln Vertical asymptote $x=2$; horizontal asymptote $y=1$. $f'(x)=-\dfrac{2}{(x-2)^2}<0$.\result{Decreasing on each side of $x=2$; it passes through the origin.}
\end{example}

\begin{example}{Example 2 \textperiodcentered{} Four shapes}
On an interval, $f'>0$ and $f''<0$. Describe the graph there.\soln Positive first derivative: rising. Negative second derivative: concave down.\result{Rising, and levelling off.}
\end{example}

\begin{example}{Example 3 \textperiodcentered{} Putting it together}
List the key features of $f(x)=x^3-3x^2$.\soln $f'(x)=3x(x-2)$: maximum $(0,0)$, minimum $(2,-4)$. $f''(x)=6x-6$: inflection $(1,-2)$. Zeros at $x=0$ and $x=3$.\result{Rises to $(0,0)$, falls through $(1,-2)$ to $(2,-4)$, then rises through $(3,0)$.}
\end{example}

\level{Core}
\begin{enumerate}[leftmargin=*,itemsep=2pt]
  \item Find the vertex and the $x$-intercepts of $f(x)=x^2-4x+3$.\work{2.3cm}
  \item Write equations for the asymptotes of $f(x)=\dfrac{2x+1}{x-3}$.\work{2.3cm}
  \item On an interval, $f'<0$ and $f''>0$. Describe the graph there.\work{2.3cm}
  \item Describe the graph of $f(x)=e^{-x}$: direction, concavity and asymptote.\work{2.3cm}
\end{enumerate}

\level{Exam level}
\begin{enumerate}[leftmargin=*,itemsep=2pt,start=5]
  \item Find the relative extrema and the $x$-coordinates of the points of inflection of $f(x)=x^4-4x^2$.\work{2.8cm}
  \item Find the maximum, the points of inflection and the horizontal asymptote of $f(x)=\dfrac{1}{x^2+1}$.\work{2.8cm}
  \item A function has $f(1)=-2$, $f'(x)<0$ for $x<1$, $f'(x)>0$ for $x>1$, and $f''(x)>0$ for all $x$. Describe its graph.\work{2.8cm}
  \item Find the relative extremum, the point of inflection and the horizontal asymptote of $f(x)=(x+1)e^{x}$.\work{2.8cm}
\end{enumerate}

\begin{frbox}{Question 9 \textperiodcentered{} Free-response practice}
Let $f(x)=\dfrac{x^2-1}{x^2+1}$.
\begin{enumerate}[label=(\alph*),leftmargin=*,itemsep=2pt]
  \item Find the horizontal asymptote and the intercepts of the graph of $f$.\work{2.6cm}
  \item Find $f'(x)$ and the coordinates of the relative extremum of $f$. Classify it.\work{2.6cm}
  \item Given that $f''(x)=\dfrac{4\left(1-3x^2\right)}{\left(x^2+1\right)^3}$, find the $x$-coordinates of the points of inflection.\work{2.6cm}
\end{enumerate}
\end{frbox}

\newpage
\begin{solbox}
\begin{enumerate}[leftmargin=*,itemsep=4pt]
  \item Vertex $(2,-1)$; intercepts $x=1$ and $x=3$.
  \item $x=3$ and $y=2$.
  \item Falling, and concave up (levelling off).
  \item Decreasing, concave up, with horizontal asymptote $y=0$ as $x\to\infty$.
  \item $f'(x)=4x\left(x^2-2\right)$: maximum $(0,0)$, minima $\left(\pm\sqrt2,-4\right)$. $f''(x)=12x^2-8$: inflection at $x=\pm\sqrt{\tfrac23}$.
  \item $f'(x)=-\dfrac{2x}{(x^2+1)^2}$: maximum $(0,1)$. $f''(x)=\dfrac{6x^2-2}{(x^2+1)^3}$: inflection at $x=\pm\dfrac{1}{\sqrt3}$. Asymptote $y=0$.
  \item It falls to a minimum at $(1,-2)$ and then rises, concave up throughout: a single bowl shape.
  \item $f'(x)=(x+2)e^{x}$: minimum $\left(-2,-e^{-2}\right)$. $f''(x)=(x+3)e^{x}$: inflection at $x=-3$. Asymptote $y=0$ as $x\to-\infty$.
  \item \begin{enumerate}[label=(\alph*),leftmargin=*,itemsep=3pt]
    \item $y=1$. Intercepts: $(0,-1)$, $(1,0)$ and $(-1,0)$.
    \item $f'(x)=\dfrac{4x}{(x^2+1)^2}$ changes from negative to positive at $x=0$: a minimum at $(0,-1)$.
    \item $f''$ changes sign where $1-3x^2=0$: $x=\pm\dfrac{1}{\sqrt3}$.
  \end{enumerate}
\end{enumerate}
\end{solbox}
