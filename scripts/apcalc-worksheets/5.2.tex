\wstitle{5.2}{Extreme Values and Critical Points}{Unit 5: Analysing Functions}

\begin{ideabox}
\begin{itemize}[leftmargin=*,itemsep=2pt,topsep=2pt]
  \item A \textbf{critical point} is a point in the domain of $f$ where $f'(x)=0$ or $f'(x)$ does not exist.
  \item Every relative extremum occurs at a critical point, but not every critical point is an extremum.
  \item \textbf{Extreme Value Theorem.} A function continuous on a \emph{closed} interval $[a,b]$ has an absolute maximum and an absolute minimum there.
  \item A point outside the domain of $f$ is never a critical point.
\end{itemize}
\end{ideabox}

\sech{Worked examples}
\begin{example}{Example 1 \textperiodcentered{} A polynomial}
Find the critical points of $f(x)=x^3-3x^2$.\soln $f'(x)=3x^2-6x=3x(x-2)$.\result{$x=0$ and $x=2$.}
\end{example}

\begin{example}{Example 2 \textperiodcentered{} Where the derivative fails}
Find the critical point of $f(x)=x^{2/3}$ and describe $f$ there.\soln $f'(x)=\dfrac23x^{-1/3}$ is never zero, and does not exist at $x=0$, which is in the domain.\result{$x=0$ is a critical point; $f$ has its minimum value 0 there.}
\end{example}

\begin{example}{Example 3 \textperiodcentered{} Using the Extreme Value Theorem}
Does $f(x)=\dfrac{1}{x-1}$ have absolute extrema on $[2,5]$? Find them.\soln $f$ is continuous on $[2,5]$, so the theorem applies. $f'(x)=-\dfrac{1}{(x-1)^2}<0$: $f$ is decreasing.\result{Maximum $f(2)=1$; minimum $f(5)=\dfrac14$.}
\end{example}

\level{Core}
\begin{enumerate}[leftmargin=*,itemsep=2pt]
  \item Find the critical point of $f(x)=x^2-10x$.\work{2.3cm}
  \item Find the critical point of $f(x)=x^4-4x$.\work{2.3cm}
  \item Find the critical points of $f(x)=\sin x$ on $(0,2\pi)$.\work{2.3cm}
  \item Does the Extreme Value Theorem apply to $f(x)=\dfrac1x$ on $[-1,1]$? Explain.\work{2.3cm}
\end{enumerate}

  \item Find all critical points of $f(x)=x-3x^{1/3}$.\work{2.8cm}
  \item Find the critical points of $f(x)=\left(x^2-4\right)^2$.\work{2.8cm}
  \item Find the critical point of $f(x)=\dfrac{\ln x}{x}$ for $x>0$, and the value of $f$ there.\work{2.8cm}
\end{enumerate}

\begin{frbox}{Question 9 \textperiodcentered{} Free-response practice}
Let $f(x)=x^4-18x^2$.
\begin{enumerate}[label=(\alph*),leftmargin=*,itemsep=2pt]
  \item Find all critical points of $f$.\work{2.6cm}
  \item Find the value of $f$ at each critical point.\work{2.6cm}
  \item Find the absolute maximum and absolute minimum values of $f$ on $[-1,4]$.\work{2.6cm}
\end{enumerate}
\end{frbox}

\newpage
\begin{solbox}
\begin{enumerate}[leftmargin=*,itemsep=4pt]
  \item $f'(x)=2x-10=0$ at $x=5$.
  \item $f'(x)=4x^3-4=0$ at $x=1$.
  \item $\cos x=0$ at $x=\dfrac\pi2$ and $x=\dfrac{3\pi}{2}$.
  \item No: $f$ is not continuous on $[-1,1]$ (it is undefined at 0).
  \item $f'(x)=e^{x/2}\left(1+\dfrac x2\right)=0$ at $x=-2$.
  \item $f'(x)=1-x^{-2/3}$: zero at $x=\pm1$, and undefined at $x=0$. Three critical points.
  \item $f'(x)=4x\left(x^2-4\right)$: $x=0$ and $x=\pm2$.
  \item $f'(x)=\dfrac{1-\ln x}{x^2}=0$ at $x=e$, where $f(e)=\dfrac1e$.
  \item \begin{enumerate}[label=(\alph*),leftmargin=*,itemsep=3pt]
    \item $f'(x)=4x^3-36x=4x(x-3)(x+3)$: $x=0$ and $x=\pm3$.
    \item $f(0)=0$ and $f(\pm3)=81-162=-81$.
    \item Candidates: $f(-1)=-17$, $f(0)=0$, $f(3)=-81$, $f(4)=-32$. Maximum 0; minimum $-81$.
  \end{enumerate}
\end{enumerate}
\end{solbox}
