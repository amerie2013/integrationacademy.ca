{\small Grade 11 Functions, University (MCR3U) \textperiodcentered{} Unit 2: Equivalent Algebraic Expressions}

\begin{ideabox}
\textbf{Factoring is expanding in reverse}: it turns a sum of terms into a product. Factors are useful because each factor $(x-a)$ gives an $x$-intercept of the graph at $x=a$, and because they let us cancel in fractions. \textbf{Every factorisation can be checked by expanding it.}
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item \textbf{Order of attack:} (1) GCF first; (2) count the terms --- 2: difference of squares; 3: trinomial; 4: grouping; (3) factor \textbf{completely}; (4) check by expanding.
  \item \textbf{Simple trinomial} $x^2+bx+c$: two numbers with product $c$ and sum $b$. \textbf{Complex trinomial} $ax^2+bx+c$: product $ac$, sum $b$; split the middle term and group.
  \item \textbf{Patterns:} $a^2-b^2=(a-b)(a+b)$ and $a^2\pm2ab+b^2=(a\pm b)^2$. A \emph{sum} of squares $a^2+b^2$ does not factor over the integers.
  \item \textbf{Quadratic form:} $x^4-13x^2+36$ is a trinomial in $u=x^2$; factor it, then factor again.
  \item \textbf{Graph link:} the factored form shows the $x$-intercepts at a glance: $y=(x-5)(x+3)$ crosses at $5$ and $-3$.
\end{itemize}
\end{ideabox}

\begin{learnbox}
\lh{Factoring undoes expanding}Factoring writes a sum as a product. Always \textbf{check by expanding}. Start with the \textbf{GCF}, then count terms: 2 terms $\to$ difference of squares; 3 terms $\to$ trinomial; 4 terms $\to$ grouping. Keep going until no factor can be factored again.
\lh{Trinomials}$x^2+bx+c$: find two numbers with product $c$ and sum $b$. $ax^2+bx+c$: find two numbers with product $ac$ and sum $b$, split the middle term, and group. If no pair works, the trinomial is prime.
\lh{Why factors matter}Each factor $(x-a)$ means the graph crosses the $x$-axis at $a$. Factors also let you cancel in rational expressions (2.3) and solve equations (the zero-product property).
\end{learnbox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Greatest common factor (two variables)}
Factor $18x^3y-24x^2y^2+6xy$.\soln Numbers: $\gcd(18,24,6)=6$; variables: $xy$ is in every term. GCF $=6xy$. \\ Divide each term: $3x^2,\ -4xy,\ 1$. \\ $=6xy(3x^2-4xy+1)$ --- note the last term leaves a \emph{1}, not nothing.
\end{example}

\begin{example}{ex1}{Example 2 --- Common binomial factor}
Factor $5x(2x+1)-3(2x+1)$.\soln The bracket $(2x+1)$ appears in both terms: take it out. \\ $=(2x + 1)(5x - 3)$ \quad (expanded: $10x^2 - x - 3$).
\end{example}

\begin{example}{ex2}{Example 3 --- Grouping}
Factor $6x^3+9x^2-10x-15$.\soln Pair the terms: $(6x^3+9x^2)+(-10x-15)=3x^2(2x+3)-5(2x+3)$. \\ The brackets match, so take out $(2x+3)$: $(2x + 3)(3x^2 - 5)$. \\ Note the $-5$: taking out $-5$ from $-10x-15$ keeps the brackets equal.
\end{example}

\begin{example}{ex3}{Example 4 --- Simple trinomial}
Factor $x^2-2x-48$.\soln Product $-48$, sum $-2$. Opposite signs; the larger number is negative: $-8$ and $6$. \\ $=(x - 8)(x + 6)$. Check: $x^2+6x-8x-48$ \checkmark.
\end{example}

\begin{example}{ex4}{Example 5 --- Complex trinomial by decomposition}
Factor $10x^2-x-21$.\soln $ac=-210$; we need product $-210$ and sum $-1$: $14$ and $-15$. \\ $10x^2+14x-15x-21=2x(5x+7)-3(5x+7)$. \\ $=(5x + 7)(2x - 3)$.
\end{example}

\begin{example}{ex5}{Example 6 --- Difference of squares and perfect squares}
Factor (a) $64x^2-49$ \quad (b) $16x^2+40x+25$.\soln (a) $(8x)^2-7^2=(8x - 7)(8x + 7)$. \\ (b) $(4x)^2$, $5^2$ and $2(4x)(5)=40x$ \checkmark, so $=(4x + 5)^2$.
\end{example}

\begin{example}{ex6}{Example 7 --- Factor completely}
Factor (a) $5x^3-45x$ \quad (b) $-3x^2+3x+36$.\soln (a) $5x(x^2-9)=5x(x - 3)(x + 3)$. \\ (b) Take out $-3$: $-3(x^2-x-12)$; product $-12$, sum $-1$: $-4,3$. $=-3(x - 4)(x + 3)$.
\end{example}

\begin{example}{ex7}{Example 8 --- Quadratic form}
Factor $x^4-5x^2+4$.\soln Let $u=x^2$: $u^2-5u+4=(u-1)(u-4)$. \\ Back-substitute: $(x^2-1)(x^2-4)=(x - 1)(x + 1)(x - 2)(x + 2)$.
\end{example}

\begin{example}{ex8}{Example 9 --- Application: when does the ball land?}
A ball's height is $h=-5t^2+15t+20$ (metres, $t$ in seconds). When does it hit the ground?\soln Factor: $h=-5(t - 4)(t + 1)$. \\ $h=0$ when $t-4=0$ or $t+1=0$: $t=4$ or $t=-1$. Time cannot be negative, so the ball lands at $t=4$ s. \\ The zeros $-1$ and $4$ are symmetric about $t=1.5$, so the maximum is at $t=1.5$ s.
\end{example}

\begin{example}{ex0}{Example 10 --- Reading a graph from the factored form}
Factor $y=x^2-2x-15$ and describe its graph.\soln $y=(x - 5)(x + 3)$: $x$-intercepts $5$ and $-3$; $y$-intercept $-15$; opens up. \begin{keybox}
\begin{multicols}{2}
\small
\begin{enumerate}[leftmargin=*,itemsep=3pt]
  \item $7x(2x - 3)$.
  \item $6ab(2ab^2-3b+1)$. Check: $6ab\cdot2ab^2=12a^2b^3$ \checkmark.
  \item $x^2(x+5)-4(x+5)=(x+5)(x^2-4)=(x + 5)(x - 2)(x + 2)$.
  \item (a) $(x + 4)(x + 7)$ \quad (b) $(x - 8)(x + 5)$. In (b) the product $-40$ is negative, so one number is positive and one negative; their sum is $-3$, so the larger in size ($-8$) is negative.
  \item $ac=72$, sum $-17$: $-8$ and $-9$. $6x^2-8x-9x+12=2x(3x-4)-3(3x-4)=(2x - 3)(3x - 4)$.
  \item $ac=-24$, sum $5$: $8$ and $-3$. $12x^2+8x-3x-2=4x(3x+2)-1(3x+2)=(4x - 1)(3x + 2)$.
  \item (a) $(9x - 4)(9x + 4)$ \quad (b) $(5x + 3)^2$.
  \item (a) $2x(x - 3)(x + 3)$ \quad (b) $(x - 1)(x + 1)(x - 4)(x + 4)$.
  \item $(x-2-6)(x-2+6)=(x - 8)(x + 4)$. Expanding: $x^2-4x+4-36=x^2-4x-32$ \checkmark.
  \item $h=-5(t - 3)(t + 1)$. Lands at $t=3$ s (the other zero $-1$ is rejected). Zeros $-1$ and $3$, so the maximum is at $t=1$: $h(1)=-5+10+15=20$ m.
  \item $x^2+14x+48=(x+6)(x+8)$: the length is $x+8$. For $x=3$: $9$ by $11$, perimeter $2(9+11)=40$.
  \item At $x=1$: $1+16=17$ but $(1+4)(1-4)=-15$, so the claim is false. For $(x+a)(x+b)=x^2+(a+b)x+ab$ we would need $a+b=0$ and $ab=16$; but $a+b=0$ means $b=-a$, so $ab=-a^2\le0$, never $16$.
  \item Intercepts: $x=-2,\ 1,\ 4$; $y$-intercept: $(2)(-1)(-4)=8$. The leading term is $x^3$, so the right end goes \emph{up} and the left end goes \emph{down}.
  \item $(x^2+1)^2-x^2=x^4+2x^2+1-x^2=x^4+x^2+1$, and it is a difference of squares: $(x^2+1-x)(x^2+1+x)=(x^2 + x + 1)(x^2 - x + 1)$.
\end{enumerate}
\end{multicols}
\end{keybox}
