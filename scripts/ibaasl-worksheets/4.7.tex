\begin{ideabox}
Z-scores standardize any normal distribution, letting probabilities and percentiles be found (or reversed).
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item $z=\dfrac{x-\mu}{\sigma}$.
  \item Inverse normal: given a probability, work backward for $x=\mu+z\sigma$.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Z-score and probability}
$\mu=80$, $\sigma=6$. Find $P(X<89)$ given $P(Z<1.5)\approx0.9332$.\soln
$z=\dfrac{89-80}{6}=1.5$.\\[3pt]\textbf{Conclusion:} $P(X<89)\approx0.9332$.
\end{example}

\begin{example}{ex1}{Example 2 --- Inverse normal}
$\mu=80$, $\sigma=6$. Find the score at the 90th percentile, given $P(Z<1.2816)\approx0.90$.\soln
$x=80+1.2816(6)$.\\[3pt]\textbf{Conclusion:} $x\approx87.69$.
\end{example}

\begin{example}{ex2}{Example 3 --- Between two values}
$\mu=80$, $\sigma=6$. Find $P(71<X<89)$ given $P(Z<-1.5)\approx0.0668$ and $P(Z<1.5)\approx0.9332$.\soln
$z_1=-1.5$, $z_2=1.5$.\\[3pt]\textbf{Conclusion:} $P\approx0.9332-0.0668=0.8664$.
\end{example}

\begin{example}{ex3}{Example 4 --- Solve for mu and sigma}
It's known $P(X<60)=0.1587$ and $P(X<100)=0.8413$. Find $\mu,\sigma$ (using $P(Z<-1)\approx0.1587$, $P(Z<1)\approx0.8413$).\soln
$\mu-\sigma=60$, $\mu+\sigma=100$. Add: $2\mu=160\Rightarrow\mu=80$; $\sigma=20$.\\[3pt]\textbf{Conclusion:} $\mu=80$, $\sigma=20$.
\end{example}
