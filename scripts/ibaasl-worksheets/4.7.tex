\begin{ideabox}
Z-scores standardize any normal distribution, letting probabilities and percentiles be found (or reversed).
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item $z=\dfrac{x-\mu}{\sigma}$.
  \item Inverse normal: given a probability, work backward for $x=\mu+z\sigma$.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Z-score and probability}
$\mu=80$, $\sigma=6$. Find $P(X<89)$ given $P(Z<1.5)\approx0.9332$.\soln
$z=\dfrac{89-80}{6}=1.5$.\\[3pt]\textbf{Conclusion:} $P(X<89)\approx0.9332$.
\end{example}

\begin{example}{ex1}{Example 2 --- Inverse normal}
$\mu=80$, $\sigma=6$. Find the score at the 90th percentile, given $P(Z<1.2816)\approx0.90$.\soln
$x=80+1.2816(6)$.\\[3pt]\textbf{Conclusion:} $x\approx87.69$.
\end{example}

\begin{example}{ex2}{Example 3 --- Between two values}
$\mu=80$, $\sigma=6$. Find $P(71<X<89)$ given $P(Z<-1.5)\approx0.0668$ and $P(Z<1.5)\approx0.9332$.\soln
$z_1=-1.5$, $z_2=1.5$.\\[3pt]\textbf{Conclusion:} $P\approx0.9332-0.0668=0.8664$.
\end{example}

\begin{example}{ex3}{Example 4 --- Solve for mu and sigma}
It's known $P(X<60)=0.1587$ and $P(X<100)=0.8413$. Find $\mu,\sigma$ (using $P(Z<-1)\approx0.1587$, $P(Z<1)\approx0.8413$).\soln
$\mu-\sigma=60$, $\mu+\sigma=100$. Add: $2\mu=160\Rightarrow\mu=80$; $\sigma=20$.\\[3pt]\textbf{Conclusion:} $\mu=80$, $\sigma=20$.
\end{example}

\begin{example}{ex4}{Example 5 --- Finding P(X > a)}
$\mu=100$, $\sigma=12$. Find $P(X>121)$ given $P(Z<1.75)\approx0.9599$.\soln
$z=\dfrac{121-100}{12}=1.75$. $P(X>121)=1-P(X<121)$.\\[3pt]\textbf{Conclusion:} $P(X>121)\approx1-0.9599=0.0401$.
\end{example}

\begin{example}{ex5}{Example 6 --- Finding a lower percentile}
$\mu=65$, $\sigma=9$. Find the value below which the lowest 15\% falls, given $P(Z<-1.0364)\approx0.15$.\soln
$x=65+(-1.0364)(9)$.\\[3pt]\textbf{Conclusion:} $x\approx55.67$.
\end{example}

\begin{example}{ex6}{Example 7 --- Sample proportion within a bound}
$\mu=500$, $\sigma=25$. A batch is rejected if a measurement is outside $[450,550]$. Find the proportion rejected, given $P(Z<-2)\approx0.0228$ and $P(Z<2)\approx0.9772$.\soln
$z_1=\dfrac{450-500}{25}=-2$, $z_2=\dfrac{550-500}{25}=2$. Accepted proportion: $0.9772-0.0228=0.9544$.\\[3pt]\textbf{Conclusion:} rejected proportion $\approx1-0.9544=0.0456$ (4.56\%).
\end{example}

\begin{example}{ex7}{Example 8 --- Solving with an asymmetric pair of z-values}
It's known $P(X<70)=0.0668$ and $P(X<130)=0.9938$ for a normal distribution. Find $\mu,\sigma$, using $P(Z<-1.5)\approx0.0668$ and $P(Z<2.5)\approx0.9938$.\soln
$\mu-1.5\sigma=70$ and $\mu+2.5\sigma=130$. Subtract: $4\sigma=60\Rightarrow\sigma=15$.\\[3pt]\textbf{Conclusion:} $\sigma=15$; $\mu=70+1.5(15)=92.5$.
\end{example}

\begin{example}{ex8}{Example 9 --- Application --- setting a pass threshold}
A company wants only the top 10\% of test scores (normal, $\mu=72$, $\sigma=8$) to qualify for advancement. Find the minimum qualifying score, given $P(Z<1.2816)\approx0.90$.\soln
The top 10\% corresponds to the 90th percentile: $x=72+1.2816(8)$.\\[3pt]\textbf{Conclusion:} minimum score $\approx82.25$, so a score of $83$ or higher would qualify.
\end{example}
