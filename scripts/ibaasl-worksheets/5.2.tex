\begin{ideabox}
Product, quotient, and chain rules combine to differentiate any layered expression.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item Product: $(uv)'=u'v+uv'$. Quotient: $\left(\dfrac uv\right)'=\dfrac{u'v-uv'}{v^2}$. Chain: $[f(g(x))]'=f'(g(x))g'(x)$.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Product rule}
Differentiate $f(x)=(3x+1)(x^2-2)$.\soln
$u=3x+1,v=x^2-2$: $f'(x)=3(x^2-2)+(3x+1)(2x)=3x^2-6+6x^2+2x$.\\[3pt]\textbf{Conclusion:} $f'(x)=9x^2+2x-6$.
\end{example}

\begin{example}{ex1}{Example 2 --- Quotient rule}
Differentiate $f(x)=\dfrac{2x}{x^2+4}$.\soln
$\dfrac{2(x^2+4)-2x(2x)}{(x^2+4)^2}=\dfrac{2x^2+8-4x^2}{(x^2+4)^2}$.\\[3pt]\textbf{Conclusion:} $f'(x)=\dfrac{8-2x^2}{(x^2+4)^2}$.
\end{example}

\begin{example}{ex2}{Example 3 --- Chain rule}
Differentiate $f(x)=(5x-2)^7$.\soln
$7(5x-2)^6\cdot5$.\\[3pt]\textbf{Conclusion:} $f'(x)=35(5x-2)^6$.
\end{example}

\begin{example}{ex3}{Example 4 --- Product combined with chain}
Differentiate $f(x)=x^3(2x+1)^4$, factoring your answer.\soln
$3x^2(2x+1)^4+x^3\cdot4(2x+1)^3\cdot2$; factor $x^2(2x+1)^3$: $3(2x+1)+8x$.\\[3pt]\textbf{Conclusion:} $f'(x)=x^2(2x+1)^3(14x+3)$.
\end{example}
