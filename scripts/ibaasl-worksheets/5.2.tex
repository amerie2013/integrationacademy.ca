\begin{ideabox}
Product, quotient, and chain rules combine to differentiate any layered expression.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item Product: $(uv)'=u'v+uv'$. Quotient: $\left(\dfrac uv\right)'=\dfrac{u'v-uv'}{v^2}$. Chain: $[f(g(x))]'=f'(g(x))g'(x)$.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Product rule}
Differentiate $f(x)=(3x+1)(x^2-2)$.\soln
$u=3x+1,v=x^2-2$: $f'(x)=3(x^2-2)+(3x+1)(2x)=3x^2-6+6x^2+2x$.\\[3pt]\textbf{Conclusion:} $f'(x)=9x^2+2x-6$.
\end{example}

\begin{example}{ex1}{Example 2 --- Quotient rule}
Differentiate $f(x)=\dfrac{2x}{x^2+4}$.\soln
$\dfrac{2(x^2+4)-2x(2x)}{(x^2+4)^2}=\dfrac{2x^2+8-4x^2}{(x^2+4)^2}$.\\[3pt]\textbf{Conclusion:} $f'(x)=\dfrac{8-2x^2}{(x^2+4)^2}$.
\end{example}

\begin{example}{ex2}{Example 3 --- Chain rule}
Differentiate $f(x)=(5x-2)^7$.\soln
$7(5x-2)^6\cdot5$.\\[3pt]\textbf{Conclusion:} $f'(x)=35(5x-2)^6$.
\end{example}

\begin{example}{ex3}{Example 4 --- Product combined with chain}
Differentiate $f(x)=x^3(2x+1)^4$, factoring your answer.\soln
$3x^2(2x+1)^4+x^3\cdot4(2x+1)^3\cdot2$; factor $x^2(2x+1)^3$: $3(2x+1)+8x$.\\[3pt]\textbf{Conclusion:} $f'(x)=x^2(2x+1)^3(14x+3)$.
\end{example}

\begin{example}{ex4}{Example 5 --- Chain rule applied twice (nested composition)}
Differentiate $f(x)=\sqrt{(3x-1)^2+4}$ (write as a power first).\soln
$f(x)=\big[(3x-1)^2+4\big]^{1/2}$; outer derivative $\tfrac12[\cdot]^{-1/2}$, inner derivative $2(3x-1)(3)=6(3x-1)$.\\[3pt]\textbf{Conclusion:} $f'(x)=\dfrac{6(3x-1)}{2\sqrt{(3x-1)^2+4}}=\dfrac{3(3x-1)}{\sqrt{(3x-1)^2+4}}$.
\end{example}

\begin{example}{ex5}{Example 6 --- Quotient rule combined with chain}
Differentiate $f(x)=\dfrac{(x+1)^2}{x-3}$.\soln
$u=(x+1)^2,v=x-3$: $u'=2(x+1)$. $f'(x)=\dfrac{2(x+1)(x-3)-(x+1)^2}{(x-3)^2}$.\\[3pt]
Factor $(x+1)$ from the numerator: $(x+1)\big[2(x-3)-(x+1)\big]=(x+1)(x-7)$.\\[3pt]\textbf{Conclusion:} $f'(x)=\dfrac{(x+1)(x-7)}{(x-3)^2}$.
\end{example}

\begin{example}{ex6}{Example 7 --- Finding where the tangent is horizontal}
Find the $x$-value(s) where the tangent to $f(x)=x^3-12x+5$ is horizontal.\soln
Horizontal tangent means $f'(x)=0$: $3x^2-12=0\Rightarrow x^2=4$.\\[3pt]\textbf{Conclusion:} $x=2$ or $x=-2$.
\end{example}

\begin{example}{ex7}{Example 8 --- Product rule with three factors (via grouping)}
Differentiate $f(x)=x(x+1)(x-2)$ by first expanding two factors.\soln
Expand $(x+1)(x-2)=x^2-x-2$, so $f(x)=x^3-x^2-2x$.\\[3pt]\textbf{Conclusion:} $f'(x)=3x^2-2x-2$.
\end{example}

\begin{example}{ex8}{Example 9 --- Application --- marginal cost}
A company's cost function is $C(x)=x^3-6x^2+400x+1000$ (dollars, for $x$ units). Find the marginal cost (the derivative) at $x=10$.\soln
$C'(x)=3x^2-12x+400$; $C'(10)=300-120+400$.\\[3pt]\textbf{Conclusion:} marginal cost at $x=10$ is \$580 per additional unit.
\end{example}
