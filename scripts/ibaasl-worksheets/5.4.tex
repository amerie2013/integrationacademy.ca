\begin{ideabox}
Critical points, the second derivative test, and inflection points reveal a curve's whole shape.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item Critical points: $f'(x)=0$; classify via a sign change (first derivative test) or $f''$'s sign (second derivative test).
  \item $f'(c)=0$ does not always mean an extremum --- check for an actual sign change.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- First derivative test}
Find and classify the critical points of $f(x)=x^3-12x$.\soln
$f'(x)=3x^2-12=3(x-2)(x+2)$; critical at $x=\pm2$. Sign: $+,-,+$ across $-2,2$.\\[3pt]\textbf{Conclusion:} local max at $x=-2$, local min at $x=2$.
\end{example}

\begin{example}{ex1}{Example 2 --- Second derivative test}
Confirm the classification using $f''$.\soln
$f''(x)=6x$; $f''(-2)=-12<0$ (max), $f''(2)=12>0$ (min).\\[3pt]\textbf{Conclusion:} confirmed.
\end{example}

\begin{example}{ex2}{Example 3 --- Inflection point}
Find the inflection point of $f(x)=x^3-12x$.\soln
$f''(x)=6x=0\Rightarrow x=0$; sign changes from $-$ to $+$. $f(0)=0$.\\[3pt]\textbf{Conclusion:} $(0,0)$.
\end{example}

\begin{example}{ex3}{Example 4 --- Critical point that is not an extremum}
Show that $f(x)=x^3+2$ has a critical point at $x=0$ that is not an extremum.\soln
$f'(x)=3x^2\geq0$ for every $x$ --- no sign change at $x=0$.\\[3pt]\textbf{Conclusion:} $x=0$ is a horizontal-tangent inflection point, not a max or min.
\end{example}

\begin{example}{ex4}{Example 5 --- Full analysis of a quartic}
Find and classify all critical points of $f(x)=x^4-8x^2$.\soln
$f'(x)=4x^3-16x=4x(x^2-4)=4x(x-2)(x+2)$; critical at $x=0,\pm2$. Sign of $f'$: $-,+,-,+$ across $-2,0,2$.\\[3pt]\textbf{Conclusion:} local min at $x=-2$, local max at $x=0$, local min at $x=2$.
\end{example}

\begin{example}{ex5}{Example 6 --- Finding both inflection points of a quartic}
Find the inflection points of $f(x)=x^4-8x^2$ (from Example 5).\soln
$f''(x)=12x^2-16=0\Rightarrow x^2=\tfrac43\Rightarrow x=\pm\sqrt{4/3}$.\\[3pt]\textbf{Conclusion:} inflection points at $x=\pm\dfrac{2}{\sqrt3}\approx\pm1.155$.
\end{example}

\begin{example}{ex6}{Example 7 --- Determining global extrema on a closed interval}
Find the global maximum and minimum of $f(x)=x^3-3x$ on $[-2,3]$.\soln
Critical points: $f'(x)=3x^2-3=0\Rightarrow x=\pm1$ (both in $[-2,3]$). Values: $f(-2)=-2$, $f(-1)=2$, $f(1)=-2$, $f(3)=18$.\\[3pt]\textbf{Conclusion:} global max $18$ at $x=3$; global min $-2$ at $x=-2$ and $x=1$ (tied).
\end{example}

\begin{example}{ex7}{Example 8 --- Curve sketching summary for a cubic}
Summarize the shape of $f(x)=-x^3+3x$ using critical points and concavity.\soln
$f'(x)=-3x^2+3=0\Rightarrow x=\pm1$; $f''(x)=-6x$: $f''(-1)=6>0$ (min), $f''(1)=-6<0$ (max). Inflection at $f''=0\Rightarrow x=0$.\\[3pt]\textbf{Conclusion:} local min at $(-1,-2)$, local max at $(1,2)$, inflection at $(0,0)$; concave up on $(-\infty,0)$, down on $(0,\infty)$.
\end{example}

\begin{example}{ex8}{Example 9 --- Why every cubic $x^3+px+q$ has exactly one inflection point}
Explain why $f(x)=x^3+px+q$ always has exactly one inflection point, for any constants $p,q$.\soln
$f''(x)=6x$, which is zero at exactly one place, $x=0$, and changes sign there (negative to positive) regardless of $p,q$.\\[3pt]\textbf{Conclusion:} every such cubic has exactly one inflection point, always at $x=0$.
\end{example}

\begin{keybox}
\begin{multicols}{2}
\small
\begin{enumerate}[leftmargin=*,itemsep=3pt]
  \item $f'=3x^2-12x=3x(x-4)$: max at $x=0$, min at $x=4$
  \item $f''=6x-12$; $f''(0)=-12<0$, $f''(4)=12>0$: confirmed
  \item $6x-12=0\Rightarrow x=2$, $f(2)=8-24=-16$: $(2,-16)$
  \item $f'=4x^3-32=4(x^3-8)=4(x-2)(x^2+2x+4)$; only real root $x=2$ (quadratic factor has no real roots); sign changes $-$ to $+$: local min at $x=2$
  \item $f'(x)=5x^4\geq0$ everywhere, no sign change: $x=0$ is a horizontal-tangent inflection, not an extremum
  \item $f'=6x^2-18x+12=6(x-1)(x-2)$: max at $x=1$, min at $x=2$; $f''=12x-18=0\Rightarrow x=1.5$ (inflection); concave down on $(-\infty,1.5)$, up on $(1.5,\infty)$
  \item $f'=4x^3-36x=4x(x^2-9)=4x(x-3)(x+3)$: min at $x=-3$, max at $x=0$, min at $x=3$
  \item $f''=12x^2-36=0\Rightarrow x^2=3\Rightarrow x=\pm\sqrt3$
  \item $f'=3x^2-12=0\Rightarrow x=\pm2$; values: $f(-3)=9$, $f(-2)=16$, $f(2)=-16$, $f(4)=16$; global max $16$ (at $x=-2$ and $x=4$, tied), global min $-16$ at $x=2$
  \item $f'=3x^2-6x=3x(x-2)$: max at $(0,0)$, min at $(2,-4)$; $f''=6x-6=0\Rightarrow x=1$ (inflection at $(1,-2)$); concave down on $(-\infty,1)$, up on $(1,\infty)$
  \item $f'=4x^3-8x=4x(x^2-2)=0\Rightarrow x=0,\pm\sqrt2$ (only $0,\sqrt2$ in $[-1,3]$); values: $f(-1)=-3$, $f(0)=0$, $f(\sqrt2)=-4$, $f(3)=45$; global max $45$ at $x=3$, global min $-4$ at $x=\sqrt2$
  \item $f''(x)=6x$, zero only at $x=0$ and changing sign there regardless of $p,q$ --- so exactly one inflection point always exists, at $x=0$
  \item $f'=15x^4-15x^2=15x^2(x^2-1)=15x^2(x-1)(x+1)$: critical at $x=0,\pm1$; sign of $f'$: $+,-,+$ crossing $-1$ and $1$ but $x=0$ has no sign change (double root): max at $x=-1$, min at $x=1$, $x=0$ is not an extremum
\end{enumerate}
\end{multicols}
\end{keybox}
