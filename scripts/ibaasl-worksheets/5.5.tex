\begin{ideabox}
Optimization reduces a two-variable problem to one variable using a constraint, then finds a critical point.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item Use the constraint to eliminate a variable, differentiate, set to 0, confirm max/min.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Maximize area with fixed fencing}
A rectangle is enclosed by 160 m of fencing. Find the dimensions maximizing area.\soln
$y=80-x$; $A=80x-x^2$; $A'=80-2x=0\Rightarrow x=40$.\\[3pt]\textbf{Conclusion:} $x=y=40$ m (a square).
\end{example}

\begin{example}{ex1}{Example 2 --- Minimize fencing with fixed area}
A pen must have area 484 m$^2$; find dimensions minimizing perimeter.\soln
$y=484/x$; $P=2x+968/x$; $P'=2-968/x^2=0\Rightarrow x^2=484\Rightarrow x=22$.\\[3pt]\textbf{Conclusion:} $x=y=22$ m.
\end{example}

\begin{example}{ex2}{Example 3 --- Open-top box}
An open box is cut from a $18\times18$ cm sheet with corner squares of side $x$. Find $x$ maximizing volume.\soln
$V=x(18-2x)^2$; $V'=(18-2x)^2+x\cdot2(18-2x)(-2)=(18-2x)(18-6x)$; zero at $x=9$ (rejected, collapses box) or $x=3$.\\[3pt]\textbf{Conclusion:} $x=3$ cm.
\end{example}

\begin{example}{ex3}{Example 4 --- Minimum surface area cylinder}
A cylinder must hold volume $432\pi$ cm$^3$; find the radius minimizing surface area.\soln
$h=432/r^2$; $S=2\pi r^2+864\pi/r$; $S'=4\pi r-864\pi/r^2=0\Rightarrow r^3=216$.\\[3pt]\textbf{Conclusion:} $r=6$ cm.
\end{example}

\begin{example}{ex4}{Example 5 --- Minimizing distance to a curve}
Find the point on the parabola $y=x^2$ closest to the point $(0,3)$.\soln
Minimize $D^2=x^2+(x^2-3)^2$ (using squared distance to avoid the root). $D^2=x^4-5x^2+9$; derivative $4x^3-10x=0\Rightarrow2x(2x^2-5)=0$.\\[3pt]\textbf{Conclusion:} $x=0$ or $x=\pm\sqrt{2.5}$; testing shows $x=\pm\sqrt{2.5}$ gives the minimum (closer than $x=0$).
\end{example}

\begin{example}{ex5}{Example 6 --- Maximizing a triangle's area with a fixed hypotenuse-style constraint}
A rectangle is inscribed under the line $x+2y=20$ in the first quadrant (one corner at the origin, opposite corner on the line). Find the dimensions maximizing its area.\soln
$y=\dfrac{20-x}2$; $A=xy=\dfrac{x(20-x)}2=10x-\dfrac{x^2}2$; $A'=10-x=0\Rightarrow x=10$.\\[3pt]\textbf{Conclusion:} $x=10$, $y=5$; max area $=50$.
\end{example}

\begin{example}{ex6}{Example 7 --- Minimizing material for an open cylinder (no lid)}
An open-top cylindrical can (no lid) must hold volume $250\pi$ cm$^3$; find the radius minimizing material used.\soln
$h=250/r^2$; $S=\pi r^2+2\pi rh=\pi r^2+500\pi/r$ (only one circular base, since it's open-top); $S'=2\pi r-500\pi/r^2=0\Rightarrow r^3=250$.\\[3pt]\textbf{Conclusion:} $r=\sqrt[3]{250}\approx6.30$ cm.
\end{example}

\begin{example}{ex7}{Example 8 --- Application --- minimizing total travel cost}
A delivery route's cost is $C(x)=2x+\dfrac{288}x$ (\$100s) for average speed-related parameter $x>0$. Find $x$ minimizing cost.\soln
$C'(x)=2-\dfrac{288}{x^2}=0\Rightarrow x^2=144$.\\[3pt]\textbf{Conclusion:} $x=12$ minimizes cost (rejecting the negative root).
\end{example}

\begin{example}{ex8}{Example 9 --- Confirming a maximum with the second derivative test}
For the rectangle-under-a-line problem (Example 6), confirm $x=10$ gives a maximum using $A''$.\soln
$A(x)=10x-\tfrac{x^2}2\Rightarrow A''(x)=-1<0$ for all $x$.\\[3pt]\textbf{Conclusion:} $A''<0$ confirms $x=10$ is indeed a maximum.
\end{example}
