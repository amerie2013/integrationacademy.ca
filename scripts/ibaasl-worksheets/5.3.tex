{\LARGE\bfseries\color{titleblue}Worksheet 5.3 --- Derivatives of Trig, Exponential and Log Functions}\\[2pt]
{\small Mathematics: Analysis \& Approaches SL (IB) \textperiodcentered{} Unit 5: Differential Calculus}
\end{center}

\begin{ideabox}
Base derivatives for sin, cos, exp, and ln extend with the chain, product, and quotient rules.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item $\dfrac{d}{dx}[\sin x]=\cos x$, $\dfrac{d}{dx}[\cos x]=-\sin x$, $\dfrac{d}{dx}[e^x]=e^x$, $\dfrac{d}{dx}[\ln x]=\dfrac1x$.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Chain rule with exponential}
Differentiate $f(x)=e^{4x^2-x}$.\soln
Inner derivative $8x-1$.\\[3pt]\textbf{Conclusion:} $f'(x)=(8x-1)e^{4x^2-x}$.
\end{example}

\begin{example}{ex1}{Example 2 --- Chain rule with logarithm}
Differentiate $f(x)=\ln(7x^2+3)$.\soln
Inner derivative $14x$.\\[3pt]\textbf{Conclusion:} $f'(x)=\dfrac{14x}{7x^2+3}$.
\end{example}

\begin{example}{ex2}{Example 3 --- Product rule with exponential}
Differentiate $f(x)=x^2e^{3x}$.\soln
$2xe^{3x}+x^2(3e^{3x})=xe^{3x}(2+3x)$.\\[3pt]\textbf{Conclusion:} $f'(x)=xe^{3x}(3x+2)$.
\end{example}

\begin{example}{ex3}{Example 4 --- Deriving sec x}
Using the quotient rule and $\sec x=\dfrac{1}{\cos x}$, prove $\dfrac{d}{dx}[\sec x]=\sec x\tan x$.\soln
$\dfrac{0\cdot\cos x-1\cdot(-\sin x)}{\cos^2x}=\dfrac{\sin x}{\cos^2x}=\dfrac1{\cos x}\cdot\dfrac{\sin x}{\cos x}$.\\[3pt]\textbf{Conclusion:} $=\sec x\tan x$.
\end{example}

\begin{example}{ex4}{Example 5 --- Chain rule with sine of a linear function}
Differentiate $f(x)=\sin(5x-2)$.\soln
Inner derivative is $5$.\\[3pt]\textbf{Conclusion:} $f'(x)=5\cos(5x-2)$.
\end{example}

\begin{example}{ex5}{Example 6 --- Chain rule nested inside a trig function}
Differentiate $f(x)=\cos(x^2+3x)$.\soln
Inner derivative is $2x+3$.\\[3pt]\textbf{Conclusion:} $f'(x)=-(2x+3)\sin(x^2+3x)$.
\end{example}

\begin{example}{ex6}{Example 7 --- Product rule combining trig and log}
Differentiate $f(x)=\sin x\ln x$.\soln
$u=\sin x,v=\ln x$: $f'(x)=\cos x\ln x+\sin x\cdot\dfrac1x$.\\[3pt]\textbf{Conclusion:} $f'(x)=\cos x\ln x+\dfrac{\sin x}{x}$.
\end{example}

\begin{example}{ex7}{Example 8 --- Quotient rule with exponential and trig}
Differentiate $f(x)=\dfrac{e^x}{\cos x}$.\soln
$\dfrac{e^x\cos x-e^x(-\sin x)}{\cos^2x}=\dfrac{e^x(\cos x+\sin x)}{\cos^2x}$.\\[3pt]\textbf{Conclusion:} $f'(x)=\dfrac{e^x(\cos x+\sin x)}{\cos^2x}$.
\end{example}

\begin{example}{ex8}{Example 9 --- Application --- rate of a decaying quantity}
A radioactive sample's mass is $M(t)=200e^{-0.05t}$ grams. Find the rate of change of mass at $t=10$.\soln
$M'(t)=200(-0.05)e^{-0.05t}=-10e^{-0.05t}$; $M'(10)=-10e^{-0.5}$.\\[3pt]\textbf{Conclusion:} $M'(10)\approx-6.07$ g/unit time --- the mass is decreasing at about 6.07 g per unit time.
\end{example}
