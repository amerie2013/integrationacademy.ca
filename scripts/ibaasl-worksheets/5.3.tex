\begin{ideabox}
Base derivatives for sin, cos, exp, and ln extend with the chain, product, and quotient rules.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item $\dfrac{d}{dx}[\sin x]=\cos x$, $\dfrac{d}{dx}[\cos x]=-\sin x$, $\dfrac{d}{dx}[e^x]=e^x$, $\dfrac{d}{dx}[\ln x]=\dfrac1x$.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Chain rule with exponential}
Differentiate $f(x)=e^{4x^2-x}$.\soln
Inner derivative $8x-1$.\\[3pt]\textbf{Conclusion:} $f'(x)=(8x-1)e^{4x^2-x}$.
\end{example}

\begin{example}{ex1}{Example 2 --- Chain rule with logarithm}
Differentiate $f(x)=\ln(7x^2+3)$.\soln
Inner derivative $14x$.\\[3pt]\textbf{Conclusion:} $f'(x)=\dfrac{14x}{7x^2+3}$.
\end{example}

\begin{example}{ex2}{Example 3 --- Product rule with exponential}
Differentiate $f(x)=x^2e^{3x}$.\soln
$2xe^{3x}+x^2(3e^{3x})=xe^{3x}(2+3x)$.\\[3pt]\textbf{Conclusion:} $f'(x)=xe^{3x}(3x+2)$.
\end{example}

\begin{example}{ex3}{Example 4 --- Deriving sec x}
Using the quotient rule and $\sec x=\dfrac{1}{\cos x}$, prove $\dfrac{d}{dx}[\sec x]=\sec x\tan x$.\soln
$\dfrac{0\cdot\cos x-1\cdot(-\sin x)}{\cos^2x}=\dfrac{\sin x}{\cos^2x}=\dfrac1{\cos x}\cdot\dfrac{\sin x}{\cos x}$.\\[3pt]\textbf{Conclusion:} $=\sec x\tan x$.
\end{example}
