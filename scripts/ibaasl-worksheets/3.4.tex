\begin{ideabox}
Radians make arc length and sector area formulas clean: $s=r\theta$, $A=\tfrac12r^2\theta$.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item $180°=\pi$ rad.
  \item Segment area $=\tfrac12r^2(\theta-\sin\theta)$.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Convert degrees to radians}
Convert $315°$ to radians as an exact multiple of $\pi$.\soln
$315\times\dfrac{\pi}{180}=\dfrac{7\pi}{4}$.\\[3pt]\textbf{Conclusion:} $\dfrac{7\pi}{4}$.
\end{example}

\begin{example}{ex1}{Example 2 --- Solve for the angle}
An arc of length 15 cm is cut from a circle of radius 6 cm. Find $\theta$ in radians.\sector{2}{100}{r}\soln
$15=6\theta\Rightarrow\theta=2.5$.\\[3pt]\textbf{Conclusion:} $\theta=2.5$ rad.
\end{example}

\begin{example}{ex2}{Example 3 --- Solve for the radius}
A sector has area 45 cm$^2$ and angle $\dfrac{3\pi}{4}$. Find $r$.\soln
$45=\tfrac12r^2\left(\tfrac{3\pi}4\right)\Rightarrow r^2=\dfrac{120}{\pi}$.\\[3pt]\textbf{Conclusion:} $r\approx6.18$ cm.
\end{example}

\begin{example}{ex3}{Example 4 --- Segment area}
Find the segment area for a circle of radius 8 cm with central angle $\dfrac{\pi}{2}$.\soln
Sector: $\tfrac12(64)(\pi/2)=16\pi$. Triangle: $\tfrac12(64)\sin(\pi/2)=32$.\\[3pt]\textbf{Conclusion:} $16\pi-32\approx18.27\text{ cm}^2$.
\end{example}

\begin{example}{ex4}{Example 5 --- Convert a non-standard radian value}
Convert $2$ radians to degrees (nearest 0.1°).\soln
$2\times\dfrac{180}{\pi}$.\\[3pt]\textbf{Conclusion:} $\approx114.6°$.
\end{example}

\begin{example}{ex5}{Example 6 --- Application --- clock hand arc}
A minute hand of length 8 cm sweeps from 12 to 4 (one third of a full circle). Find the arc length traced by its tip.\soln
One third of a circle is $\dfrac{2\pi}3$ rad. $s=8\left(\dfrac{2\pi}3\right)$.\\[3pt]\textbf{Conclusion:} $s=\dfrac{16\pi}3\approx16.76$ cm.
\end{example}

\begin{example}{ex6}{Example 7 --- Application --- pizza slice area}
A pizza of radius 15 cm is cut into a slice with central angle $50°$. Find the slice's area.\soln
Convert: $\theta=50\times\dfrac{\pi}{180}=\dfrac{5\pi}{18}\approx0.8727$ rad. Area $=\tfrac12(225)(0.8727)$.\\[3pt]\textbf{Conclusion:} area $\approx98.18\text{ cm}^2$.
\end{example}

\begin{example}{ex7}{Example 8 --- Application --- circular window segment}
A circular window of radius 40 cm has a chord cutting off a segment with central angle $\dfrac{2\pi}5$. Find the segment's area (nearest cm$^2$).\soln
Sector: $\tfrac12(1600)\left(\tfrac{2\pi}5\right)=320\pi\approx1005.3$. Triangle: $\tfrac12(1600)\sin\!\left(\tfrac{2\pi}5\right)\approx760.9$.\\[3pt]\textbf{Conclusion:} segment area $\approx244\text{ cm}^2$.
\end{example}

\begin{example}{ex8}{Example 9 --- Sector area from arc length and radius}
A sector's arc length is 20 cm and its radius is 8 cm. Find the sector's area.\soln
$\theta=\dfrac{20}8=2.5$ rad. Area $=\tfrac12(64)(2.5)$.\\[3pt]\textbf{Conclusion:} area $=80\text{ cm}^2$.
\end{example}

  \item $60=\tfrac12(100)\theta\Rightarrow\theta=1.2$ rad
  \item sector $=\tfrac12(144)(2\pi/3)=48\pi$; triangle $=\tfrac12(144)\sin(2\pi/3)=72\cdot\tfrac{\sqrt3}2\approx62.35$; segment $\approx150.80-62.35\approx88.45\text{ cm}^2$
  \item $2r+r\theta=30$, $\tfrac12r^2\theta=54\Rightarrow r\theta=108/r$; $2r+108/r=30\Rightarrow2r^2-30r+108=0\Rightarrow r^2-15r+54=0\Rightarrow(r-6)(r-9)=0$: $r=6$ or $r=9$
  \item straight sides $2(90)=180$; two semicircles $=2(12\pi)=24\pi$; total $\approx180+75.4=255.4$ m

  \item sector $=\tfrac12(900)(\pi/3)=150\pi\approx471.24$; triangle $=\tfrac12(900)\sin(\pi/3)\approx389.71$; segment $\approx82\text{ cm}^2$

\end{enumerate}
