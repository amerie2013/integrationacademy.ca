\begin{ideabox}
Integration reverses differentiation; every indefinite integral carries an unknown constant $C$.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item $\displaystyle\int x^n\,dx=\dfrac{x^{n+1}}{n+1}+C$.
  \item $\displaystyle\int(ax+b)^n\,dx=\dfrac{(ax+b)^{n+1}}{a(n+1)}+C$.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Power rule}
Find $\displaystyle\int(12x^3-4x+7)\,dx$.\soln
$3x^4-2x^2+7x+C$.\\[3pt]\textbf{Conclusion:} $3x^4-2x^2+7x+C$.
\end{example}

\begin{example}{ex1}{Example 2 --- Finding C}
Given $f'(x)=4x-3$, $f(2)=5$, find $f(x)$.\soln
$f(x)=2x^2-3x+C$; $5=8-6+C\Rightarrow C=3$.\\[3pt]\textbf{Conclusion:} $f(x)=2x^2-3x+3$.
\end{example}

\begin{example}{ex2}{Example 3 --- Linear substitution}
Find $\displaystyle\int(4x-1)^5\,dx$.\soln
Divide by $a=4$: $\dfrac{(4x-1)^6}{24}$.\\[3pt]\textbf{Conclusion:} $\dfrac{(4x-1)^6}{24}+C$.
\end{example}

\begin{example}{ex3}{Example 4 --- From acceleration to position}
$a(t)=8t-6$, $v(0)=2$, $s(0)=-3$. Find $s(t)$.\soln
$v(t)=4t^2-6t+2$ (using $v(0)=2$). $s(t)=\tfrac43t^3-3t^2+2t-3$ (using $s(0)=-3$).\\[3pt]\textbf{Conclusion:} $s(t)=\tfrac43t^3-3t^2+2t-3$.
\end{example}

\begin{example}{ex4}{Example 5 --- Rewriting a root before integrating}
Find $\displaystyle\int\sqrt x\,dx$ by rewriting as a fractional power.\soln
$\displaystyle\int x^{1/2}\,dx=\dfrac{x^{3/2}}{3/2}$.\\[3pt]\textbf{Conclusion:} $\dfrac23x^{3/2}+C$.
\end{example}

\begin{example}{ex5}{Example 6 --- Splitting a fraction with two terms}
Find $\displaystyle\int\dfrac{6x^2+4x}{2x}\,dx$.\soln
Split termwise: $\dfrac{6x^2}{2x}+\dfrac{4x}{2x}=3x+2$.\\[3pt]\textbf{Conclusion:} $\dfrac{3x^2}2+2x+C$.
\end{example}

\begin{example}{ex6}{Example 7 --- Finding a curve given its slope and a point}
A curve has $\dfrac{dy}{dx}=6x^2-4x+1$ and passes through $(1,3)$. Find $y$.\soln
$y=2x^3-2x^2+x+C$; substitute $(1,3)$: $3=2-2+1+C\Rightarrow C=2$.\\[3pt]\textbf{Conclusion:} $y=2x^3-2x^2+x+2$.
\end{example}

\begin{example}{ex7}{Example 8 --- Two-stage integration with two conditions}
Given $f''(x)=12x$, $f'(0)=3$, $f(0)=-1$, find $f(x)$.\soln
$f'(x)=6x^2+3$ (using $f'(0)=3$). $f(x)=2x^3+3x-1$ (using $f(0)=-1$).\\[3pt]\textbf{Conclusion:} $f(x)=2x^3+3x-1$.
\end{example}

\begin{example}{ex8}{Example 9 --- Application --- reconstructing distance from acceleration}
A car accelerates according to $a(t)=2$ m/s$^2$ starting from rest ($v(0)=0$) at position $s(0)=0$. Find its position after $t=6$ s.\soln
$v(t)=2t$; $s(t)=t^2$.\\[3pt]\textbf{Conclusion:} $s(6)=36$ m.
\end{example}

  \item $3x^5-4x^2+2x+C$
  \item $f(x)=2x^3+x+C$; $4=2+1+C\Rightarrow C=1$: $f(x)=2x^3+x+1$
  \item $\dfrac{(2x+7)^5}{10}+C$
  \item simplify to $3x^2-2$: $x^3-2x+C$
  \item $v(t)=3t^2-4t$; $s(t)=t^3-2t^2+2$
  \item $\dfrac{(3x-2)^4}{12}+C$; differentiating gives $\dfrac{4(3x-2)^3\cdot3}{12}=(3x-2)^3$, matching the original integrand
  \item $\int x^{1/3}dx=\dfrac{x^{4/3}}{4/3}=\dfrac34x^{4/3}+C$
  \item simplify to $2x^2-3$: $\dfrac{2x^3}3-3x+C$
  \item $y=x^3-3x^2+2x+C$; $1=8-12+4+C\Rightarrow C=1$: $y=x^3-3x^2+2x+1$
  \item $f'(x)=3x^2-4x+C_1$; $1=3-4+C_1\Rightarrow C_1=2$; $f(x)=x^3-2x^2+2x+C_2$; $2=0+C_2\Rightarrow C_2=2$: $f(x)=x^3-2x^2+2x+2$
  \item $v(t)=4t$; $s(t)=2t^2$; $s(5)=50$ m
  \item expand to $x^2+x-6$: $\dfrac{x^3}3+\dfrac{x^2}2-6x+C$
  \item $\int(5x+2)^{-2}dx=\dfrac{(5x+2)^{-1}}{5(-1)}+C=-\dfrac1{5(5x+2)}+C$
\end{enumerate}
