{\LARGE\bfseries\color{titleblue}Worksheet 2.2 --- Transformations of Functions}\\[2pt]
{\small Mathematics: Analysis \& Approaches SL (IB) \textperiodcentered{} Unit 2: Functions}
\end{center}

\begin{ideabox}
Every transformed graph is a parent function shifted, stretched, or reflected.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item $y=f(x)+k$ shifts vertically; $y=f(x+k)$ shifts horizontally (opposite the sign).
  \item $y=af(x)$ stretches vertically by $a$; $y=f(bx)$ stretches horizontally by $\tfrac1b$.
  \item $y=-f(x)$ reflects in the $x$-axis; $y=f(-x)$ reflects in the $y$-axis.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Describe a transformation}
Describe the transformation from $y=x^2$ to $y=(x+4)^2-6$.\soln
Shift left 4, down 6; new vertex $(-4,-6)$.\\[3pt]\textbf{Conclusion:} left 4, down 6.
\end{example}

\begin{example}{ex1}{Example 2 --- Vertical stretch and reflection}
Describe $y=-3x^2$ as a transformation of $y=x^2$.\soln
Factor $3$ stretches vertically; negative sign reflects in the $x$-axis.\\[3pt]\textbf{Conclusion:} vertically stretched by 3, reflected in the x-axis.
\end{example}

\begin{example}{ex2}{Example 3 --- Root function shift}
Describe the transformation from $y=\sqrt x$ to $y=\sqrt{x-5}+2$.\soln
Right 5, up 2.\\[3pt]\textbf{Conclusion:} right 5, up 2.
\end{example}

\begin{example}{ex3}{Example 4 --- Horizontal compression}
Given $f(x)=x^2$, describe $y=f(3x)$.\soln
$y=(3x)^2=9x^2$ --- horizontal compression by $\tfrac13$ (equivalently vertical stretch by 9).\\[3pt]\textbf{Conclusion:} compressed horizontally by $\tfrac13$.
\end{example}

\begin{example}{ex4}{Example 5 --- Combining a stretch, shift, and reflection}
Describe $y=-2(x-3)^2+5$ as transformations of $y=x^2$, in order.\soln
Stretch vertically by 2, reflect in the $x$-axis, then shift right 3 and up 5.\\[3pt]\textbf{Conclusion:} reflected, stretched by 2, shifted right 3 up 5; vertex $(3,5)$.
\end{example}

\begin{example}{ex5}{Example 6 --- Transformation of a root function with compression}
Given $f(x)=\sqrt x$, describe $y=f(4x)-3$.\soln
$y=\sqrt{4x}-3=2\sqrt x-3$ --- equivalent to a vertical stretch by 2 combined with a shift down 3.\\[3pt]\textbf{Conclusion:} vertical stretch by 2 (from the horizontal compression), shifted down 3.
\end{example}

\begin{example}{ex6}{Example 7 --- Tracking an intercept through transformations}
A graph $y=f(x)$ has an $x$-intercept at $x=5$. Find the $x$-intercept of $y=f(2x-4)$ (factor the input first: $f(2(x-2))$).\soln
The transformation is a horizontal compression by $\tfrac12$ then a shift right 2. The original root at $x=5$ maps: first compress ($x=2.5$), then shift right 2.\\[3pt]\textbf{Conclusion:} new root at $x=4.5$.
\end{example}

\begin{example}{ex7}{Example 8 --- Application --- modelling a shifted signal}
A sensor's baseline reading is $y=f(t)$. A second, recalibrated sensor reads $3$ units higher and lags behind by $2$ seconds. Write its model in terms of $f$.\soln
"Lags behind by 2 seconds" means the same reading happens 2 seconds later --- a shift right by 2. "3 units higher" is a vertical shift up 3.\\[3pt]\textbf{Conclusion:} the second sensor's model is $y=f(t-2)+3$.
\end{example}

\begin{example}{ex8}{Example 9 --- Reversing a transformation to find the original function}
A graph $y=2f(x+1)-4$ has a maximum at $(3,6)$. Find the coordinates of the corresponding maximum on $y=f(x)$.\soln
\textbf{Step 1:} Undo the vertical stretch/shift: original $y$-value satisfies $6=2y-4\Rightarrow y=5$.\\[3pt]
\textbf{Step 2:} Undo the horizontal shift: $x+1=3\Rightarrow x=2$.\\[3pt]\textbf{Conclusion:} the maximum on $y=f(x)$ is at $(2,5)$. ✓
\end{example}
