\begin{ideabox}
Solving trig equations combines the unit circle's symmetry with ordinary algebra --- including factoring quadratics in $\sin x$ or $\cos x$.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item $y=a\sin(b(x-c))+d$: amplitude $|a|$, period $\dfrac{2\pi}{b}$, midline $y=d$.
  \item Quadratic-in-trig equations factor exactly like ordinary quadratics after substituting $u=\sin x$ or $u=\cos x$.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- State transformation features}
State the amplitude, period, and midline of $y=6\cos(3x)-2$.\splot{-9}{5}{\addplot[exblue,thick]{6*cos(deg(3*x))-2};}\soln
Amplitude 6, period $\dfrac{2\pi}{3}$, midline $y=-2$.\\[3pt]\textbf{Conclusion:} amplitude 6, period $2\pi/3$, midline $y=-2$.
\end{example}

\begin{example}{ex1}{Example 2 --- Solve a linear trig equation}
Solve $\cos x=-0.4$ for $0\leq x\leq2\pi$.\soln
Reference angle $\cos^{-1}(0.4)\approx1.159$. Q2: $x\approx\pi-1.159\approx1.983$. Q3: $x\approx\pi+1.159\approx4.301$.\\[3pt]\textbf{Conclusion:} $x\approx1.983,4.301$.
\end{example}

\begin{example}{ex2}{Example 3 --- Quadratic in sine}
Solve $2\sin^2x+\sin x-1=0$ for $0\leq x\leq2\pi$.\soln
$(2\sin x-1)(\sin x+1)=0$. $\sin x=\tfrac12\Rightarrow x=\tfrac{\pi}6,\tfrac{5\pi}6$. $\sin x=-1\Rightarrow x=\tfrac{3\pi}2$.\\[3pt]\textbf{Conclusion:} $x=\tfrac{\pi}6,\tfrac{5\pi}6,\tfrac{3\pi}2$.
\end{example}

\begin{example}{ex3}{Example 4 --- Identity substitution first}
Solve $2\cos^2x-\sin x-1=0$ for $0\leq x\leq2\pi$.\soln
$2(1-\sin^2x)-\sin x-1=0\Rightarrow-2\sin^2x-\sin x+1=0\Rightarrow2\sin^2x+\sin x-1=0\Rightarrow(2\sin x-1)(\sin x+1)=0$.\\[3pt]\textbf{Conclusion:} $x=\tfrac{\pi}6,\tfrac{5\pi}6,\tfrac{3\pi}2$ (same equation as Example 3, after substitution).
\end{example}

\end{enumerate}
