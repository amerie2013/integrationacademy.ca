\begin{ideabox}
Solving trig equations combines the unit circle's symmetry with ordinary algebra --- including factoring quadratics in $\sin x$ or $\cos x$.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item $y=a\sin(b(x-c))+d$: amplitude $|a|$, period $\dfrac{2\pi}{b}$, midline $y=d$.
  \item Quadratic-in-trig equations factor exactly like ordinary quadratics after substituting $u=\sin x$ or $u=\cos x$.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- State transformation features}
State the amplitude, period, and midline of $y=6\cos(3x)-2$.\splot{-9}{5}{\addplot[exblue,thick]{6*cos(deg(3*x))-2};}\soln
Amplitude 6, period $\dfrac{2\pi}{3}$, midline $y=-2$.\\[3pt]\textbf{Conclusion:} amplitude 6, period $2\pi/3$, midline $y=-2$.
\end{example}

\begin{example}{ex1}{Example 2 --- Solve a linear trig equation}
Solve $\cos x=-0.4$ for $0\leq x\leq2\pi$.\soln
Reference angle $\cos^{-1}(0.4)\approx1.159$. Q2: $x\approx\pi-1.159\approx1.983$. Q3: $x\approx\pi+1.159\approx4.301$.\\[3pt]\textbf{Conclusion:} $x\approx1.983,4.301$.
\end{example}

\begin{example}{ex2}{Example 3 --- Quadratic in sine}
Solve $2\sin^2x+\sin x-1=0$ for $0\leq x\leq2\pi$.\soln
$(2\sin x-1)(\sin x+1)=0$. $\sin x=\tfrac12\Rightarrow x=\tfrac{\pi}6,\tfrac{5\pi}6$. $\sin x=-1\Rightarrow x=\tfrac{3\pi}2$.\\[3pt]\textbf{Conclusion:} $x=\tfrac{\pi}6,\tfrac{5\pi}6,\tfrac{3\pi}2$.
\end{example}

\begin{example}{ex3}{Example 4 --- Identity substitution first}
Solve $2\cos^2x-\sin x-1=0$ for $0\leq x\leq2\pi$.\soln
$2(1-\sin^2x)-\sin x-1=0\Rightarrow-2\sin^2x-\sin x+1=0\Rightarrow2\sin^2x+\sin x-1=0\Rightarrow(2\sin x-1)(\sin x+1)=0$.\\[3pt]\textbf{Conclusion:} $x=\tfrac{\pi}6,\tfrac{5\pi}6,\tfrac{3\pi}2$ (same equation as Example 3, after substitution).
\end{example}

\begin{example}{ex4}{Example 5 --- Extended domain (more solutions)}
Solve $\sin x=0.5$ for $0\leq x\leq4\pi$.\soln
In $[0,2\pi]$: $x=\tfrac{\pi}6,\tfrac{5\pi}6$. Add $2\pi$ for the second cycle: $x=\tfrac{13\pi}6,\tfrac{17\pi}6$.\\[3pt]\textbf{Conclusion:} $x=\tfrac{\pi}6,\tfrac{5\pi}6,\tfrac{13\pi}6,\tfrac{17\pi}6$.
\end{example}

\begin{example}{ex5}{Example 6 --- Quadratic-in-cosine requiring the quadratic formula}
Solve $3\cos^2x-\cos x-1=0$ for $0\leq x\leq2\pi$ (this doesn't factor nicely).\soln
Let $u=\cos x$: $u=\dfrac{1\pm\sqrt{1+12}}6=\dfrac{1\pm\sqrt{13}}6$, giving $u\approx0.768$ or $u\approx-0.434$.\\[3pt]
\textbf{Conclusion:} $x\approx0.696,\,2.009,\,4.275,\,5.588$ (four solutions, from the two cosine values each giving two angles).
\end{example}

\begin{example}{ex6}{Example 7 --- Factoring out a common trig factor}
Solve $2\sin x\cos x-\sin x=0$ for $0\leq x\leq2\pi$.\soln
Factor: $\sin x(2\cos x-1)=0$. $\sin x=0\Rightarrow x=0,\pi,2\pi$. $\cos x=\tfrac12\Rightarrow x=\tfrac{\pi}3,\tfrac{5\pi}3$.\\[3pt]\textbf{Conclusion:} $x=0,\tfrac{\pi}3,\pi,\tfrac{5\pi}3,2\pi$.
\end{example}

\begin{example}{ex7}{Example 8 --- Finding a maximum and when it first occurs}
A wave's height is $h(t)=5\sin(2t)+3$; find the maximum height and the first time $t>0$ it's reached.\soln
Maximum of $\sin$ is 1, giving $h=5+3=8$, at $2t=\tfrac{\pi}2$.\\[3pt]\textbf{Conclusion:} max height $8$, first reached at $t=\tfrac{\pi}4$.
\end{example}

\begin{example}{ex8}{Example 9 --- Application --- bridge clearance}
A bridge's vertical clearance is $C(t)=2\cos\!\left(\dfrac{\pi}6t\right)+5$ m, for $t\in[0,24)$ hours. Find all times the clearance equals 4 m.\soln
$2\cos(\pi t/6)=-1\Rightarrow\cos(\pi t/6)=-0.5\Rightarrow\pi t/6=\tfrac{2\pi}3$ or $\tfrac{4\pi}3$ (period is 12 h, so this repeats once more in $[0,24)$).\\[3pt]\textbf{Conclusion:} $t=4,\,8,\,16,\,20$ hours.
\end{example}
