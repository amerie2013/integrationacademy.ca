\begin{ideabox}
A geometric sequence has a constant common ratio $r$ between consecutive terms.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item General term: $u_n=u_1r^{n-1}$.
  \item Sum of the first $n$ terms: $S_n=\dfrac{u_1(r^n-1)}{r-1}$, $r\neq1$.
  \item Sum to infinity (only when $|r|<1$): $S_\infty=\dfrac{u_1}{1-r}$.
\end{itemize}
\end{ideabox}

\begin{learnbox}
\lh{When does $S_\infty$ exist?}Only when $-1<r<1$ --- otherwise the terms don't shrink toward 0 and the sum grows without bound.
\lh{Finding $r$ from two terms}Divide two given terms to eliminate $u_1$: $\dfrac{u_m}{u_n}=r^{m-n}$, then solve for $r$.
\end{learnbox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Find a term}
Find $u_9$ for $3,6,12,\dots$\soln
$u_1=3$, $r=2$. $u_9=3(2)^8=3(256)$.\\[3pt]\textbf{Conclusion:} $u_9=768$.
\end{example}

\begin{example}{ex1}{Example 2 --- Find the common ratio from two terms}
A geometric sequence has $u_1=500$ and $u_6=15.625$. Find $r$ (assume $r>0$).\soln
$u_6=u_1r^5\Rightarrow15.625=500r^5\Rightarrow r^5=0.03125=\tfrac{1}{32}$.\\[3pt]\textbf{Conclusion:} $r=\tfrac12$.
\end{example}

\begin{example}{ex2}{Example 3 --- Sum to infinity}
Find $S_\infty$ for $40,-20,10,-5,\dots$\soln
$u_1=40$, $r=-\tfrac12$ (valid, since $|r|<1$). $S_\infty=\dfrac{40}{1-(-1/2)}=\dfrac{40}{3/2}$.\\[3pt]\textbf{Conclusion:} $S_\infty=\dfrac{80}{3}$.
\end{example}

\begin{example}{ex3}{Example 4 --- Application --- depreciation}
A machine worth \$18{,}000 loses 20\% of its value each year. Find its value after 5 years, and the total depreciation over that time.\soln
\textbf{Step 1:} Value follows $u_n=18000(0.8)^n$; after 5 years, $u_5=18000(0.8)^5\approx5898.24$.\\[3pt]
\textbf{Step 2:} Total depreciation $=18000-5898.24$.\\[3pt]
\textbf{Conclusion:} value $\approx\$5898.24$; depreciation $\approx\$12{,}101.76$.
\end{example}

\begin{example}{ex4}{Example 5 --- Solve for $u\_1$ given $S\_infty$}
A geometric series has $r=0.4$ and $S_\infty=60$. Find $u_1$.\soln
$60=\dfrac{u_1}{1-0.4}=\dfrac{u_1}{0.6}$.\\[3pt]\textbf{Conclusion:} $u_1=36$.
\end{example}

\begin{example}{ex5}{Example 6 --- Application --- population growth}
A town of 5000 grows by 3\% each year. Find its population after 8 years (nearest whole number).\soln
$u_9=5000(1.03)^8$ (year 8 growth applied 8 times from the start).\\[3pt]\textbf{Conclusion:} population $\approx6333$.
\end{example}

\begin{example}{ex6}{Example 7 --- Sum of a finite geometric series with negative $r$}
Find $S_7$ for $2,-6,18,-54,\dots$\soln
$u_1=2$, $r=-3$. $S_7=\dfrac{2\big((-3)^7-1\big)}{-3-1}=\dfrac{2(-2187-1)}{-4}$.\\[3pt]\textbf{Conclusion:} $S_7=\dfrac{2(-2188)}{-4}=1094$.
\end{example}

\begin{example}{ex7}{Example 8 --- Solve for $n$ given a term value}
A geometric sequence has $u_1=5$, $r=2$. Find $n$ such that $u_n=1280$.\soln
$5(2)^{n-1}=1280\Rightarrow2^{n-1}=256=2^8$.\\[3pt]\textbf{Conclusion:} $n-1=8\Rightarrow n=9$. ✓
\end{example}

\begin{example}{ex8}{Example 9 --- Application --- doubling investment}
An investment doubles every 6 years. If it starts at \$2000, express its value after $t$ periods of 6 years as a geometric sequence, and find its value after 4 periods (24 years).\soln
\textbf{Step 1:} $u_1=2000$, $r=2$ per period. $u_5=2000(2)^4$ (value after 4 periods is the 5th term, counting the start as term 1).\\[3pt]
\textbf{Conclusion:} $u_5=2000(16)=\$32{,}000$. ✓
\end{example}
