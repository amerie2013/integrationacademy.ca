{\small Mathematics: Analysis \& Approaches SL (IB) \textperiodcentered{} Unit 1: Number and Algebra}
\end{center}

\begin{ideabox}
Logarithms are the inverse of exponentials: $\log_b(x)=y\iff b^y=x$.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item $\log_b(mn)=\log_bm+\log_bn$, $\log_b\!\left(\dfrac mn\right)=\log_bm-\log_bn$, $\log_b(m^k)=k\log_bm$.
  \item Change of base: $\log_ba=\dfrac{\ln a}{\ln b}$.
  \item Use logarithms to solve exponential equations with unmatched bases.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Evaluate directly}
Evaluate $\log_3(81)$.\soln
$3^{?}=81=3^4$.\\[3pt]\textbf{Conclusion:} $4$.
\end{example}

\begin{example}{ex1}{Example 2 --- Solve via logarithms}
Solve $6^x=50$ to 3 decimal places.\soln
Take $\ln$ of both sides: $x\ln6=\ln50\Rightarrow x=\dfrac{\ln50}{\ln6}$.\\[3pt]\textbf{Conclusion:} $x\approx2.184$.
\end{example}

\begin{example}{ex2}{Example 3 --- Simplify using log laws}
Simplify $\log_2(20)+\log_2(4)-\log_2(5)$.\soln
$=\log_2\!\left(\dfrac{20\times4}{5}\right)=\log_2(16)$.\\[3pt]\textbf{Conclusion:} $4$.
\end{example}

\begin{example}{ex3}{Example 4 --- Solve with a domain check}
Solve $\log_3(x)+\log_3(x-2)=1$, checking the domain.\soln
\textbf{Step 1:} $\log_3\big(x(x-2)\big)=1\Rightarrow x(x-2)=3\Rightarrow x^2-2x-3=0\Rightarrow(x-3)(x+1)=0$.\\[3pt]
\textbf{Step 2:} $x=3$ or $x=-1$; but $\log_3(x)$ requires $x>0$, so $x=-1$ is rejected.\\[3pt]
\textbf{Conclusion:} $x=3$.
\end{example}

\begin{example}{ex4}{Example 5 --- Change of base}
Evaluate $\log_7(50)$ to 3 decimal places using change of base.\soln
$\log_7(50)=\dfrac{\ln50}{\ln7}$.\\[3pt]\textbf{Conclusion:} $\approx2.010$.
\end{example}

\begin{example}{ex5}{Example 6 --- Solve for an exponent using change of base}
Solve $3^{x+1}=20$ to 3 decimal places.\soln
Take $\ln$: $(x+1)\ln3=\ln20\Rightarrow x+1=\dfrac{\ln20}{\ln3}$.\\[3pt]\textbf{Conclusion:} $x\approx1.727$.
\end{example}

\begin{example}{ex6}{Example 7 --- Application --- half-life}
A substance decays according to $A=A_0(0.5)^{t/8}$ (8-day half-life). Find, to the nearest day, when only 20\% remains.\soln
$0.2=0.5^{t/8}\Rightarrow\ln0.2=\dfrac t8\ln0.5\Rightarrow t=8\cdot\dfrac{\ln0.2}{\ln0.5}$.\\[3pt]\textbf{Conclusion:} $t\approx19$ days.
\end{example}

\begin{example}{ex7}{Example 8 --- Combine multiple log laws}
Simplify $3\log_2(x)-\log_2(x^2)+\log_2(4)$ as a single logarithm.\soln
$=\log_2(x^3)-\log_2(x^2)+\log_2(4)=\log_2\!\left(\dfrac{4x^3}{x^2}\right)$.\\[3pt]\textbf{Conclusion:} $\log_2(4x)$.
\end{example}

\begin{example}{ex8}{Example 9 --- Solve a log equation reducing to a quadratic}
Solve $(\log_2x)^2-3\log_2x+2=0$ (let $u=\log_2x$).\soln
$u^2-3u+2=0\Rightarrow(u-1)(u-2)=0\Rightarrow u=1$ or $u=2$.\\[3pt]
\textbf{Conclusion:} $\log_2x=1\Rightarrow x=2$; $\log_2x=2\Rightarrow x=4$. ✓
\end{example}
