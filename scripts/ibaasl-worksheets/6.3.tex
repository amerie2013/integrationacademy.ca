\begin{ideabox}
Reverse the trig, exponential, and log derivatives, including the linear-substitution pattern.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item $\displaystyle\int\sin x\,dx=-\cos x+C$, $\int\cos x\,dx=\sin x+C$, $\int e^x\,dx=e^x+C$, $\int\tfrac1x\,dx=\ln|x|+C$.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Basic combination}
Find $\displaystyle\int(5\cos x+3e^x)\,dx$.\soln
$5\sin x+3e^x+C$.\\[3pt]\textbf{Conclusion:} $5\sin x+3e^x+C$.
\end{example}

\begin{example}{ex1}{Example 2 --- Exponential substitution}
Find $\displaystyle\int e^{2x+7}\,dx$.\soln
Divide by $2$: $\dfrac{e^{2x+7}}{2}$.\\[3pt]\textbf{Conclusion:} $\dfrac{e^{2x+7}}{2}+C$.
\end{example}

\begin{example}{ex2}{Example 3 --- Trig substitution}
Find $\displaystyle\int\cos(3x-1)\,dx$.\soln
Divide by $3$: $\dfrac{\sin(3x-1)}{3}$.\\[3pt]\textbf{Conclusion:} $\dfrac{\sin(3x-1)}{3}+C$.
\end{example}

\begin{example}{ex3}{Example 4 --- Definite integral, combined types}
Evaluate $\displaystyle\int_0^{\pi/2}(3\sin x+\cos x)\,dx$ exactly.\soln
Antiderivative $-3\cos x+\sin x$; at $\pi/2$: $0+1=1$; at $0$: $-3+0=-3$.\\[3pt]\textbf{Conclusion:} $1-(-3)=4$.
\end{example}
