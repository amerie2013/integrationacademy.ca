\begin{ideabox}
A definite integral gives net signed area; true area requires splitting at every x-intercept.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item $\displaystyle\int_a^bf(x)\,dx=F(b)-F(a)$.
  \item If $f$ changes sign in $[a,b]$, split at the intercept and take absolute values before adding.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Basic evaluation}
Evaluate $\displaystyle\int_1^4(3x+2)\,dx$.\soln
$\left[\tfrac32x^2+2x\right]_1^4=(24+8)-(1.5+2)$.\\[3pt]\textbf{Conclusion:} $32-3.5=28.5$.
\end{example}

\begin{example}{ex1}{Example 2 --- Area under a curve}
Find the area under $f(x)=x^2$ from $x=1$ to $x=4$.\soln
$\left[\tfrac{x^3}3\right]_1^4=\tfrac{64}3-\tfrac13$.\\[3pt]\textbf{Conclusion:} $21$ square units.
\end{example}

\begin{example}{ex2}{Example 3 --- Curve dips below the axis}
Find the total area between $f(x)=x^2-9$ and the $x$-axis on $[0,4]$.\soln
Intercept at $x=3$. $\int_0^3(x^2-9)dx=9-27=-18$; $\int_3^4(x^2-9)dx=(64/3-36)-(9-27)=10/3$.\\[3pt]\textbf{Conclusion:} total $=18+10/3=64/3$.
\end{example}

\begin{example}{ex3}{Example 4 --- Integral properties}
Given $\displaystyle\int_0^8f\,dx=25$ and $\int_0^5f\,dx=14$, find $\int_5^8f\,dx$.\soln
$25-14$.\\[3pt]\textbf{Conclusion:} $11$.
\end{example}

\begin{example}{ex4}{Example 5 --- Evaluating with a negative lower limit}
Evaluate $\displaystyle\int_{-2}^{3}(2x+1)\,dx$.\soln
$[x^2+x]_{-2}^3=(9+3)-(4-2)$.\\[3pt]\textbf{Conclusion:} $12-2=10$.
\end{example}

\begin{example}{ex5}{Example 6 --- Area under a curve entirely below the axis}
Find the true area between $f(x)=-x^2-1$ and the $x$-axis on $[0,3]$ (note $f<0$ throughout).\soln
$\int_0^3(-x^2-1)dx=\left[-\tfrac{x^3}3-x\right]_0^3=-9-3=-12$ (a negative signed area, since the curve never crosses the axis here).\\[3pt]\textbf{Conclusion:} true area $=|-12|=12$ square units.
\end{example}

\begin{example}{ex6}{Example 7 --- Reversing the limits of integration}
Given $\displaystyle\int_2^7f(x)\,dx=15$, find $\displaystyle\int_7^2f(x)\,dx$.\soln
Swapping the limits negates the integral: $\int_7^2f\,dx=-\int_2^7f\,dx$.\\[3pt]\textbf{Conclusion:} $-15$.
\end{example}

\begin{example}{ex7}{Example 8 --- Definite integral requiring linear substitution first}
Evaluate $\displaystyle\int_1^3(2x-1)^2\,dx$.\soln
Antiderivative: $\dfrac{(2x-1)^3}{6}$. At $x=3$: $\dfrac{125}6$. At $x=1$: $\dfrac16$.\\[3pt]\textbf{Conclusion:} $\dfrac{125}6-\dfrac16=\dfrac{124}6=\dfrac{62}3$.
\end{example}

\begin{example}{ex8}{Example 9 --- Application --- accumulated rainfall}
A rainfall rate is $r(t)=4-0.5t$ mm/hr for $0\leq t\leq10$ hours. Find the total rainfall, and explain the meaning of the point where $r(t)=0$.\soln
$\int_0^{10}(4-0.5t)dt=[4t-0.25t^2]_0^{10}=40-25=15$ mm.\\[3pt]\textbf{Conclusion:} total rainfall $15$ mm; $r(t)=0$ at $t=8$, meaning it stops raining at that point (rate becomes negative afterward, which isn't physically meaningful for rainfall, so the model likely only applies up to $t=8$).
\end{example}
