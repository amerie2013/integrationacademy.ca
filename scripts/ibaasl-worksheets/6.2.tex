\begin{ideabox}
A definite integral gives net signed area; true area requires splitting at every x-intercept.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item $\displaystyle\int_a^bf(x)\,dx=F(b)-F(a)$.
  \item If $f$ changes sign in $[a,b]$, split at the intercept and take absolute values before adding.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Basic evaluation}
Evaluate $\displaystyle\int_1^4(3x+2)\,dx$.\soln
$\left[\tfrac32x^2+2x\right]_1^4=(24+8)-(1.5+2)$.\\[3pt]\textbf{Conclusion:} $32-3.5=28.5$.
\end{example}

\begin{example}{ex1}{Example 2 --- Area under a curve}
Find the area under $f(x)=x^2$ from $x=1$ to $x=4$.\soln
$\left[\tfrac{x^3}3\right]_1^4=\tfrac{64}3-\tfrac13$.\\[3pt]\textbf{Conclusion:} $21$ square units.
\end{example}

\begin{example}{ex2}{Example 3 --- Curve dips below the axis}
Find the total area between $f(x)=x^2-9$ and the $x$-axis on $[0,4]$.\soln
Intercept at $x=3$. $\int_0^3(x^2-9)dx=9-27=-18$; $\int_3^4(x^2-9)dx=(64/3-36)-(9-27)=10/3$.\\[3pt]\textbf{Conclusion:} total $=18+10/3=64/3$.
\end{example}

\begin{example}{ex3}{Example 4 --- Integral properties}
Given $\displaystyle\int_0^8f\,dx=25$ and $\int_0^5f\,dx=14$, find $\int_5^8f\,dx$.\soln
$25-14$.\\[3pt]\textbf{Conclusion:} $11$.
\end{example}

\begin{keybox}
\begin{multicols}{2}
\small
\begin{enumerate}[leftmargin=*,itemsep=3pt]
  \item $[2x^2+x]_0^3=18+3=21$
  \item $[16x-x^3/3]_{-4}^4=(64-64/3)-(-64+64/3)=\tfrac{256}3$
  \item intercepts $0,4$; $\int_0^4(x^2-4x)dx=[\tfrac{x^3}3-2x^2]_0^4=\tfrac{64}3-32=-\tfrac{32}3$; $\int_4^5(x^2-4x)dx=(\tfrac{125}3-50)-(\tfrac{64}3-32)=\tfrac{61}3-18=\tfrac73$; total $=\tfrac{32}3+\tfrac73=13$
  \item $40-22=18$
  \item $[12t-2t^2]_0^5=60-50=10$ L
  \item intercepts $x=-1,3$; $\int_{-2}^{-1}(x^2-2x-3)dx=(-\tfrac13-1+3)-(-\tfrac83-4+6)=\tfrac53-(-\tfrac23)=\tfrac73$; $\int_{-1}^3(\cdot)dx=(9-9-9)-(-\tfrac13-1+3)=-9-\tfrac53=-\tfrac{32}3$; $\int_3^4(\cdot)dx=(\tfrac{64}3-16-12)-(9-9-9)=-\tfrac83+9=\tfrac{19}3$; total $=\tfrac73+\tfrac{32}3+\tfrac{19}3=\tfrac{58}3$
\end{enumerate}
\end{multicols}
\end{keybox}
