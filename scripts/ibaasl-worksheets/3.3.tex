{\LARGE\bfseries\color{titleblue}Worksheet 3.3 --- The Sine Rule and Cosine Rule}\\[2pt]
{\small Mathematics: Analysis \& Approaches SL (IB) \textperiodcentered{} Unit 3: Geometry and Trigonometry}
\end{center}

\begin{ideabox}
The sine and cosine rules solve any triangle, including the tricky ambiguous case.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item $\dfrac{a}{\sin A}=\dfrac{b}{\sin B}=\dfrac{c}{\sin C}$; $a^2=b^2+c^2-2bc\cos A$.
  \item Area $=\tfrac12ab\sin C$.
  \item Ambiguous case (SSA): check whether both $\theta$ and $180°-\theta$ give valid triangles.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Sine rule for a side}
In $\triangle ABC$, $A=35°$, $B=70°$, $a=14$. Find $b$.\gtri{A}{B}{C}{a}{b}{c}\soln
$\dfrac{14}{\sin35°}=\dfrac{b}{\sin70°}\Rightarrow b=\dfrac{14\sin70°}{\sin35°}$.\\[3pt]\textbf{Conclusion:} $b\approx22.9$.
\end{example}

\begin{example}{ex1}{Example 2 --- Cosine rule for the largest angle}
A triangle has sides $6,8,11$. Find the largest angle.\soln
Opposite $11$: $\cos C=\dfrac{36+64-121}{96}=\dfrac{-21}{96}$.\\[3pt]\textbf{Conclusion:} $C\approx102.6°$.
\end{example}

\begin{example}{ex2}{Example 3 --- Ambiguous case}
In $\triangle ABC$, $A=40°$, $a=9$, $b=13$. Find both possible values of $B$.\soln
$\sin B=\dfrac{13\sin40°}{9}\approx0.928$. $B_1\approx68.2°$; $B_2=180°-68.2°=111.8°$; check $A+B_2=151.8°<180°$, valid.\\[3pt]\textbf{Conclusion:} $B\approx68.2°$ or $111.8°$.
\end{example}

\begin{example}{ex3}{Example 4 --- Navigation with the cosine rule}
A ship sails 30 km on a bearing of $050°$, then 45 km on a bearing of $140°$. Find the direct distance home.\soln
Included angle $=140-50=90°$. $d^2=900+2025-0=2925$.\\[3pt]\textbf{Conclusion:} $d\approx54.1$ km.
\end{example}

\begin{example}{ex4}{Example 5 --- Cosine rule for a side, obtuse angle}
In $\triangle ABC$, $b=10$, $c=14$, $A=110°$. Find $a$.\soln
$a^2=100+196-2(10)(14)\cos110°=296-280(-0.342)$.\\[3pt]\textbf{Conclusion:} $a^2\approx391.8\Rightarrow a\approx19.79$.
\end{example}

\begin{example}{ex5}{Example 6 --- Ambiguous case with only one valid triangle}
In $\triangle ABC$, $A=110°$, $a=20$, $b=15$. Find $B$, and explain why only one triangle is possible.\soln
$\sin B=\dfrac{15\sin110°}{20}\approx0.7048\Rightarrow B_1\approx44.8°$. The supplementary option $B_2=135.2°$ would make $A+B_2>180°$, impossible.\\[3pt]\textbf{Conclusion:} only $B\approx44.8°$ is valid --- an obtuse given angle always rules out the second case.
\end{example}

\begin{example}{ex6}{Example 7 --- Area via the cosine rule first}
Find the area of a triangle with sides $8,10,13$.\soln
Find the angle opposite 13: $\cos C=\dfrac{64+100-169}{160}=-0.03125\Rightarrow C\approx91.8°$.\\[3pt]\textbf{Conclusion:} Area $=\tfrac12(8)(10)\sin(91.8°)\approx39.98$ square units.
\end{example}

\begin{example}{ex7}{Example 8 --- Two-leg navigation distance}
A plane flies 40 km on a bearing of $015°$, then 55 km on a bearing of $100°$. Find the distance between the start and end points.\soln
Included angle $=100-15=85°$. $d^2=1600+3025-2(40)(55)\cos85°$.\\[3pt]\textbf{Conclusion:} $d^2\approx4241.3\Rightarrow d\approx65.1$ km.
\end{example}

\begin{example}{ex8}{Example 9 --- Application --- cost from a triangular area}
A triangular garden has sides 18 m and 24 m with an included angle of $55°$. Sod costs \$8 per m$^2$. Find the total cost.\soln
Area $=\tfrac12(18)(24)\sin55°\approx176.9\text{ m}^2$.\\[3pt]\textbf{Conclusion:} cost $\approx176.9\times8\approx\$1415.20$.
\end{example}
