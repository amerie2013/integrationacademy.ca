{\small Mathematics: Analysis \& Approaches SL (IB) \textperiodcentered{} Unit 1: Number and Algebra}
\end{center}

\begin{ideabox}
The Binomial Theorem expands $(a+b)^n$ without multiplying it out term by term.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item $(a+b)^n=\displaystyle\sum_{r=0}^{n}\binom nr a^{n-r}b^r$.
  \item General term: $T_{r+1}=\dbinom nr a^{n-r}b^r$.
  \item $\dbinom nr=\dfrac{n!}{r!(n-r)!}$ --- Pascal's triangle row $n$.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Full expansion}
Expand $(x+2)^4$ fully.\soln
$\displaystyle\sum_{r=0}^{4}\binom4r x^{4-r}2^r=x^4+4x^3(2)+6x^2(4)+4x(8)+16$.\\[3pt]\textbf{Conclusion:} $x^4+8x^3+24x^2+32x+16$.
\end{example}

\begin{example}{ex1}{Example 2 --- One coefficient}
Find the coefficient of $x^5$ in the expansion of $(x+3)^8$.\soln
$T_{r+1}=\binom8r x^{8-r}3^r$; need $8-r=5\Rightarrow r=3$. $\binom83 3^3=56(27)$.\\[3pt]\textbf{Conclusion:} coefficient $=1512$.
\end{example}

\begin{example}{ex2}{Example 3 --- Constant term}
Find the constant term in the expansion of $\left(x^2+\dfrac{2}{x}\right)^6$.\soln
$T_{r+1}=\binom6r (x^2)^{6-r}\left(\dfrac2x\right)^r=\binom6r 2^r x^{12-3r}$; need $12-3r=0\Rightarrow r=4$. $\binom64 2^4=15(16)$.\\[3pt]\textbf{Conclusion:} constant term $=240$.
\end{example}

\begin{example}{ex3}{Example 4 --- Approximation}
Use the first three terms of the binomial expansion of $(1+0.01)^5$ to approximate $1.01^5$ to 4 decimal places.\soln
$(1+0.01)^5\approx1+5(0.01)+10(0.01)^2=1+0.05+0.001$.\\[3pt]\textbf{Conclusion:} $1.01^5\approx1.0510$.
\end{example}
