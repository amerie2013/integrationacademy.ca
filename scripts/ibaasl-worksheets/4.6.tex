\begin{ideabox}
The binomial distribution models a fixed number of independent yes/no trials.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item $E(X)=\sum xP(X=x)$.
  \item $X\sim B(n,p)$: $P(X=k)=\dbinom nk p^k(1-p)^{n-k}$, $E(X)=np$, $\text{Var}(X)=np(1-p)$.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Unknown table entry}
$P(X=0)=0.1$, $P(X=1)=k$, $P(X=2)=0.5$. Find $k$ and $E(X)$.\soln
$k=1-0.1-0.5=0.4$. $E(X)=0(0.1)+1(0.4)+2(0.5)$.\\[3pt]\textbf{Conclusion:} $k=0.4$, $E(X)=1.4$.
\end{example}

\begin{example}{ex1}{Example 2 --- Binomial probability}
$X\sim B(9,0.35)$. Find $P(X=3)$.\soln
$\binom93(0.35)^3(0.65)^6\approx84(0.042875)(0.0754)\approx0.272$.\\[3pt]\textbf{Conclusion:} $P(X=3)\approx0.272$.
\end{example}

\begin{example}{ex2}{Example 3 --- At least one}
$X\sim B(6,0.15)$. Find $P(X\geq1)$.\soln
$P(X=0)=(0.85)^6\approx0.377$.\\[3pt]\textbf{Conclusion:} $P(X\geq1)\approx0.623$.
\end{example}

\begin{example}{ex3}{Example 4 --- Solve for p}
$X\sim B(5,p)$ and $P(X=5)=0.00243$. Find $p$.\soln
$p^5=0.00243\Rightarrow p=0.00243^{1/5}=0.3$ (recognizing $0.3^5=0.00243$).\\[3pt]\textbf{Conclusion:} $p=0.3$.
\end{example}

\begin{example}{ex4}{Example 5 --- Expected value of a custom discrete variable}
A game pays \$10 with probability 0.2, \$0 with probability 0.5, and $-\$5$ (a loss) with probability 0.3. Find the expected value.\soln
$E(X)=10(0.2)+0(0.5)+(-5)(0.3)=2+0-1.5$.\\[3pt]\textbf{Conclusion:} $E(X)=\$0.50$ --- the game is favourable on average.
\end{example}

\begin{example}{ex5}{Example 6 --- At most $k$ successes}
$X\sim B(6,0.25)$. Find $P(X\leq1)$.\soln
$P(X=0)=(0.75)^6\approx0.178$. $P(X=1)=\binom61(0.25)(0.75)^5\approx0.356$.\\[3pt]\textbf{Conclusion:} $P(X\leq1)\approx0.178+0.356=0.534$.
\end{example}

\begin{example}{ex6}{Example 7 --- Between two values}
$X\sim B(10,0.4)$. Find $P(2\leq X\leq4)$.\soln
$P(2)=\binom{10}2(0.4)^2(0.6)^8\approx0.121$; $P(3)\approx0.215$; $P(4)\approx0.251$.\\[3pt]\textbf{Conclusion:} $P(2\leq X\leq4)\approx0.121+0.215+0.251=0.587$.
\end{example}

\begin{example}{ex7}{Example 8 --- Solve for $n$ given a mean}
$X\sim B(n,0.3)$ and $E(X)=12$. Find $n$.\soln
$np=12\Rightarrow0.3n=12$.\\[3pt]\textbf{Conclusion:} $n=40$.
\end{example}

\begin{example}{ex8}{Example 9 --- Application --- warranty claims}
A store sells 20 items with a 5\% individual chance of a warranty claim. Find the expected number of claims and the probability of at least 2 claims.\soln
$E(X)=20(0.05)=1$. $P(X\geq2)=1-P(0)-P(1)=1-(0.95)^{20}-20(0.05)(0.95)^{19}\approx1-0.358-0.377$.\\[3pt]\textbf{Conclusion:} expected claims $=1$; $P(X\geq2)\approx0.265$.
\end{example}

  \item $(1-p)^4=0.0016\Rightarrow1-p=0.0016^{1/4}=0.2\Rightarrow p=0.8$
  \item $P(0)=(0.96)^{10}\approx0.6648$; $P(1)=\binom{10}1(0.04)(0.96)^9\approx0.2770$; sum $\approx0.9418$
