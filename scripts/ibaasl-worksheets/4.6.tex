\begin{ideabox}
The binomial distribution models a fixed number of independent yes/no trials.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item $E(X)=\sum xP(X=x)$.
  \item $X\sim B(n,p)$: $P(X=k)=\dbinom nk p^k(1-p)^{n-k}$, $E(X)=np$, $\text{Var}(X)=np(1-p)$.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Unknown table entry}
$P(X=0)=0.1$, $P(X=1)=k$, $P(X=2)=0.5$. Find $k$ and $E(X)$.\soln
$k=1-0.1-0.5=0.4$. $E(X)=0(0.1)+1(0.4)+2(0.5)$.\\[3pt]\textbf{Conclusion:} $k=0.4$, $E(X)=1.4$.
\end{example}

\begin{example}{ex1}{Example 2 --- Binomial probability}
$X\sim B(9,0.35)$. Find $P(X=3)$.\soln
$\binom93(0.35)^3(0.65)^6\approx84(0.042875)(0.0754)\approx0.272$.\\[3pt]\textbf{Conclusion:} $P(X=3)\approx0.272$.
\end{example}

\begin{example}{ex2}{Example 3 --- At least one}
$X\sim B(6,0.15)$. Find $P(X\geq1)$.\soln
$P(X=0)=(0.85)^6\approx0.377$.\\[3pt]\textbf{Conclusion:} $P(X\geq1)\approx0.623$.
\end{example}

\begin{example}{ex3}{Example 4 --- Solve for p}
$X\sim B(5,p)$ and $P(X=5)=0.00243$. Find $p$.\soln
$p^5=0.00243\Rightarrow p=0.00243^{1/5}=0.3$ (recognizing $0.3^5=0.00243$).\\[3pt]\textbf{Conclusion:} $p=0.3$.
\end{example}
