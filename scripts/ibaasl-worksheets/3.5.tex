{\LARGE\bfseries\color{titleblue}Worksheet 3.5 --- The Unit Circle and Trigonometric Identities}\\[2pt]
{\small Mathematics: Analysis \& Approaches SL (IB) \textperiodcentered{} Unit 3: Geometry and Trigonometry}

\begin{ideabox}
The unit circle and the Pythagorean identity let you find exact trig values in any quadrant.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item $\sin^2\theta+\cos^2\theta=1$, $\tan\theta=\dfrac{\sin\theta}{\cos\theta}$.
  \item Quadrant signs (All-Students-Take-Calculus): Q1 all $+$, Q2 sin $+$, Q3 tan $+$, Q4 cos $+$.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Exact value}
Find the exact value of $\sin(300°)$.\soln
Reference angle $60°$, Q4 (sin negative): $-\sin(60°)=-\dfrac{\sqrt3}{2}$.\\[3pt]\textbf{Conclusion:} $-\dfrac{\sqrt3}2$.
\end{example}

\begin{example}{ex1}{Example 2 --- Quadrant and Pythagorean identity}
Given $\sin\theta=-\dfrac{5}{13}$, $\theta$ in Q3, find $\cos\theta$.\soln
$\cos^2\theta=1-\tfrac{25}{169}=\tfrac{144}{169}$; Q3 cosine negative.\\[3pt]\textbf{Conclusion:} $\cos\theta=-\dfrac{12}{13}$.
\end{example}

\begin{example}{ex2}{Example 3 --- Prove an identity}
Prove $(\sin x+\cos x)^2=1+2\sin x\cos x$.\soln
Expand: $\sin^2x+2\sin x\cos x+\cos^2x=1+2\sin x\cos x$ (using the Pythagorean identity).\\[3pt]\textbf{Conclusion:} identity confirmed.
\end{example}

\begin{example}{ex3}{Example 4 --- From tangent to sine and cosine}
Given $\tan\theta=-\dfrac34$, $\theta$ in Q2, find $\sin\theta$ and $\cos\theta$.\soln
Reference triangle: opp 3, adj 4, hyp 5. Q2: sin $+$, cos $-$.\\[3pt]\textbf{Conclusion:} $\sin\theta=\dfrac35$, $\cos\theta=-\dfrac45$.
\end{example}
