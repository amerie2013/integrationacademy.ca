{\LARGE\bfseries\color{titleblue}Worksheet 3.1 --- 3D Geometry: Volume, Surface Area and Distance}\\[2pt]
{\small Mathematics: Analysis \& Approaches SL (IB) \textperiodcentered{} Unit 3: Geometry and Trigonometry}

\begin{ideabox}
3D distance and volume formulas extend familiar 2D ideas one dimension further.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item Distance: $d=\sqrt{(x_2-x_1)^2+(y_2-y_1)^2+(z_2-z_1)^2}$.
  \item Cone: $V=\tfrac13\pi r^2h$, curved area $\pi rl$. Sphere: $V=\tfrac43\pi r^3$, area $4\pi r^2$.
  \item Angle between a diagonal and a face: find the base diagonal first, then use right-triangle trig with the height.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- 3D distance}
Find the distance between $P(2,-1,4)$ and $Q(6,3,1)$.\soln
$d=\sqrt{16+16+9}=\sqrt{41}$.\\[3pt]\textbf{Conclusion:} $d=\sqrt{41}\approx6.40$.
\end{example}

\begin{example}{ex1}{Example 2 --- Composite solid surface area}
A cone (radius 5 m, height 12 m) is topped with a hemisphere of the same radius. Find the total exposed curved area.\soln
Slant height $l=\sqrt{25+144}=13$. Cone: $\pi(5)(13)=65\pi$. Hemisphere: $\tfrac12(4\pi\cdot25)=50\pi$.\\[3pt]\textbf{Conclusion:} total $=115\pi\approx361.3\text{ m}^2$.
\end{example}

\begin{example}{ex2}{Example 3 --- Pyramid height}
A square pyramid has base side 9 cm and volume 189 cm$^3$. Find its height.\soln
Base area $=81$; $189=\tfrac13(81)h=27h$.\\[3pt]\textbf{Conclusion:} $h=7$ cm.
\end{example}

\begin{example}{ex3}{Example 4 --- Angle with the base}
A box has length 12 cm, width 5 cm, height 9 cm. Find the angle the space diagonal makes with the base.\soln
Base diagonal $=\sqrt{144+25}=13$. $\tan\theta=\dfrac{9}{13}\Rightarrow\theta=\tan^{-1}(0.6923)$.\\[3pt]\textbf{Conclusion:} $\theta\approx34.7°$.
\end{example}

\begin{example}{ex4}{Example 5 --- Midpoint in 3D}
Find the midpoint of $A(3,-5,7)$ and $B(-1,9,1)$.\soln
Average each coordinate: $\left(\dfrac{3-1}2,\dfrac{-5+9}2,\dfrac{7+1}2\right)$.\\[3pt]\textbf{Conclusion:} midpoint $=(1,2,4)$.
\end{example}

\begin{example}{ex5}{Example 6 --- Composite solid volume}
An ice-cream-cone shape has a cone of radius 3 cm, height 8 cm, topped with a hemisphere of the same radius. Find the total volume.\soln
Cone: $\tfrac13\pi(9)(8)=24\pi$. Hemisphere: $\tfrac12\left(\tfrac43\pi(27)\right)=18\pi$.\\[3pt]\textbf{Conclusion:} total volume $=42\pi\approx131.9\text{ cm}^3$.
\end{example}

\begin{example}{ex6}{Example 7 --- Angle between a base diagonal and an edge}
A box's base has length 6 cm and width 8 cm. Find the angle the base's diagonal makes with the 6 cm edge.\soln
$\tan\theta=\dfrac{8}{6}=\dfrac43\Rightarrow\theta=\tan^{-1}(1.333)$.\\[3pt]\textbf{Conclusion:} $\theta\approx53.1°$.
\end{example}

\begin{example}{ex7}{Example 8 --- Solve for an unknown dimension}
A box has length $x$, width $x+2$, height 5 cm, and volume 175 cm$^3$. Find $x$.\soln
$5x(x+2)=175\Rightarrow x(x+2)=35\Rightarrow x^2+2x-35=0\Rightarrow(x-5)(x+7)=0$.\\[3pt]\textbf{Conclusion:} $x=5$ cm (rejecting the negative root).
\end{example}

\begin{example}{ex8}{Example 9 --- Application --- surface to paint}
A cylindrical silo (radius 3 m, height 10 m) has a hemispherical roof of the same radius. Find the total area to paint, excluding the base (which sits on the ground).\soln
Cylinder curved: $2\pi(3)(10)=60\pi$. Hemisphere: $2\pi(9)=18\pi$.\\[3pt]\textbf{Conclusion:} total $=78\pi\approx245.0\text{ m}^2$.
\end{example}
