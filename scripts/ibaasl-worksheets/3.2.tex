\begin{ideabox}
Multi-step right-triangle problems often chain two triangles that share a common side.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item $\sin\theta=\dfrac{\text{opp}}{\text{hyp}}$, $\cos\theta=\dfrac{\text{adj}}{\text{hyp}}$, $\tan\theta=\dfrac{\text{opp}}{\text{adj}}$.
  \item Two given angles of elevation from different positions usually mean two right triangles sharing the unknown height.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Angle of elevation with eye height}
A person whose eyes are 1.7 m up stands 25 m from a tower and sees its top at an elevation angle of $32°$. Find the tower's height.\rtri{4}{2.4}{opp}{25 m}{}\soln
$x=25\tan(32°)\approx15.6$ m; add eye height.\\[3pt]\textbf{Conclusion:} height $\approx17.3$ m.
\end{example}

\begin{example}{ex1}{Example 2 --- Two-step angle of depression}
From a 60 m cliff, the angle of depression to a boat is $22°$. It sails away until the angle of depression is $11°$. Find the distance sailed.\soln
$d_1=\dfrac{60}{\tan22°}\approx148.5$; $d_2=\dfrac{60}{\tan11°}\approx308.6$.\\[3pt]\textbf{Conclusion:} distance $\approx160.1$ m.
\end{example}

\begin{example}{ex2}{Example 3 --- Flagpole on a building}
A 5 m flagpole sits atop a building. From the ground, the angle of elevation to the pole's base is $38°$, and to its top is $44°$. Find the building's height.\soln
$h\tan(44°)=(h+5)\tan(38°)$ combined via $d$; solving $h=\dfrac{5\tan(38°)}{\tan(44°)-\tan(38°)}$.\\[3pt]\textbf{Conclusion:} $h\approx18.5$ m.
\end{example}

\begin{example}{ex3}{Example 4 --- Bearing components}
A hiker walks 10 km on a bearing of $055°$. Find how far north and east she travels.\soln
North: $10\cos55°\approx5.74$; East: $10\sin55°\approx8.19$.\\[3pt]\textbf{Conclusion:} $\approx5.74$ km north, $\approx8.19$ km east.
\end{example}
