\begin{ideabox}
Vertex form reveals a parabola's turning point immediately.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item Line through two points: $m=\dfrac{y_2-y_1}{x_2-x_1}$, then $y-y_1=m(x-x_1)$.
  \item Vertex form: $y=a(x-h)^2+k$, vertex $(h,k)$.
  \item Complete the square to convert $y=ax^2+bx+c$ into vertex form.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Equation of a line}
Find the equation of the line through $(-1,3)$ and $(4,18)$.\soln
$m=\dfrac{18-3}{4-(-1)}=\dfrac{15}{5}=3$. $y-3=3(x+1)\Rightarrow y=3x+6$.\\[3pt]\textbf{Conclusion:} $y=3x+6$.
\end{example}

\begin{example}{ex1}{Example 2 --- Complete the square}
Convert $y=x^2-10x+18$ to vertex form.\soln
Half of $-10$ squared is $25$: $y=(x^2-10x+25)-25+18=(x-5)^2-7$.\\[3pt]\textbf{Conclusion:} $y=(x-5)^2-7$, vertex $(5,-7)$.
\end{example}

\begin{example}{ex2}{Example 3 --- Vertex form when $a
eq1$}
Convert $y=2x^2+8x+3$ to vertex form.\soln
Factor 2 from the first two terms: $y=2(x^2+4x)+3=2(x^2+4x+4-4)+3=2(x+2)^2-8+3$.\\[3pt]\textbf{Conclusion:} $y=2(x+2)^2-5$.
\end{example}

\begin{example}{ex3}{Example 4 --- Build a quadratic from vertex and a point}
Find the equation of a parabola with vertex $(-2,5)$ through $(1,-13)$.\soln
$y=a(x+2)^2+5$. Substitute $(1,-13)$: $-13=a(9)+5\Rightarrow a=-2$.\\[3pt]\textbf{Conclusion:} $y=-2(x+2)^2+5$.
\end{example}
