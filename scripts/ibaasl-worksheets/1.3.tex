{\LARGE\bfseries\color{titleblue}Worksheet 1.3 --- Sigma Notation}\\[2pt]
{\small Mathematics: Analysis \& Approaches SL (IB) \textperiodcentered{} Unit 1: Number and Algebra}
\end{center}

\begin{ideabox}
Sigma notation $\displaystyle\sum_{n=1}^{N}f(n)$ compactly writes the sum of $f(n)$ from $n=1$ to $n=N$.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item $\displaystyle\sum_{n=a}^{b}f(n)$: substitute each integer $n$ from $a$ to $b$ into $f(n)$ and add.
  \item If $f(n)$ is linear in $n$, the sum is an arithmetic series; if $f(n)$ involves $r^n$, it's geometric.
  \item Splitting or shifting the index (e.g.\ starting at $n=0$ vs $n=1$) can change the formula without changing the sum's value.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Evaluate directly}
Evaluate $\displaystyle\sum_{n=1}^{5}(3n-2)$.\soln
Terms: $1,4,7,10,13$.\\[3pt]\textbf{Conclusion:} sum $=35$.
\end{example}

\begin{example}{ex1}{Example 2 --- Geometric sigma sum}
Evaluate $\displaystyle\sum_{n=1}^{6}4(2)^{n-1}$.\soln
This is a geometric series with $u_1=4$, $r=2$, $n=6$ terms: $S_6=\dfrac{4(2^6-1)}{2-1}=4(63)$.\\[3pt]\textbf{Conclusion:} $252$.
\end{example}

\begin{example}{ex2}{Example 3 --- Write a series in sigma form}
Write $8+13+18+\cdots+68$ in sigma notation and evaluate it.\soln
\textbf{Step 1:} Arithmetic, $u_1=8$, $d=5$, general term $u_n=8+5(n-1)=5n+3$.\\[3pt]
\textbf{Step 2:} Find $n$: $5n+3=68\Rightarrow n=13$. So $\displaystyle\sum_{n=1}^{13}(5n+3)$.\\[3pt]
\textbf{Step 3:} $S_{13}=\dfrac{13}{2}(8+68)=\dfrac{13}{2}(76)$.\\[3pt]\textbf{Conclusion:} sigma form $\sum_{n=1}^{13}(5n+3)$, value $=494$.
\end{example}

\begin{example}{ex3}{Example 4 --- Application --- total savings}
A savings plan deposits $D_n=25+3n$ dollars in month $n$, for $n=1$ to $12$. Find the total deposited over the year.\soln
$\displaystyle\sum_{n=1}^{12}(25+3n)=12(25)+3\sum_{n=1}^{12}n=300+3\left(\dfrac{12(13)}{2}\right)=300+3(78)$.\\[3pt]\textbf{Conclusion:} total $=\$534$.
\end{example}

\begin{example}{ex4}{Example 5 --- Splitting a sum into known pieces}
Evaluate $\displaystyle\sum_{n=1}^{10}(4n^2-3n)$ using $\sum n=\tfrac{n(n+1)}2$ and $\sum n^2=\tfrac{n(n+1)(2n+1)}6$.\soln
$\sum_{n=1}^{10}n^2=\tfrac{10(11)(21)}6=385$; $\sum_{n=1}^{10}n=55$.\\[3pt]\textbf{Conclusion:} $4(385)-3(55)=1540-165=1375$.
\end{example}

\begin{example}{ex5}{Example 6 --- Shifted-index geometric sum}
Evaluate $\displaystyle\sum_{n=0}^{5}3(2)^{n}$.\soln
This is geometric with $u_1=3$ (at $n=0$), $r=2$, and $6$ terms (from $n=0$ to $5$): $S_6=\dfrac{3(2^6-1)}{1}$.\\[3pt]\textbf{Conclusion:} $S_6=3(63)=189$.
\end{example}

\begin{example}{ex6}{Example 7 --- Application --- attendance decline}
A festival's attendance is modelled by $A_n=2000-40n$ for day $n=1$ to $10$. Find the total attendance over the 10 days.\soln
$\displaystyle\sum_{n=1}^{10}(2000-40n)=10(2000)-40\sum_{n=1}^{10}n=20000-40(55)$.\\[3pt]\textbf{Conclusion:} total $=20000-2200=17800$ attendees.
\end{example}

\begin{example}{ex7}{Example 8 --- Find an unknown upper limit}
Find $N$ such that $\displaystyle\sum_{n=1}^{N}(2n-1)=144$.\soln
\textbf{Step 1:} This is arithmetic ($u_1=1,d=2$): $S_N=\dfrac N2(2+2(N-1))=N^2$.\\[3pt]
\textbf{Conclusion:} $N^2=144\Rightarrow N=12$. ✓
\end{example}

\begin{example}{ex8}{Example 9 --- Sigma sum requiring factoring first}
Evaluate $\displaystyle\sum_{n=1}^{5}n(n+1)$ by expanding first.\soln
$n(n+1)=n^2+n$, so the sum is $\sum n^2+\sum n=\tfrac{5(6)(11)}6+\tfrac{5(6)}2=55+15$.\\[3pt]\textbf{Conclusion:} $70$.
\end{example}

  \item terms $3,5,\dots,15$: sum $=63$
  \item $S_5=\tfrac{5(3^5-1)}{2}=605$
  \item $8+15+24+35=82$
  \item $u_n=5n-1$, $n=13$; $\sum_{n=1}^{13}(5n-1)$, $S_{13}=\tfrac{13}{2}(4+64)=442$
  \item $u_n=2(3)^{n-1}$, $n=6$; $\sum_{n=1}^{6}2(3)^{n-1}$, $S_6=\tfrac{2(3^6-1)}{2}=728$
  \item $S=\tfrac{7(1-0.5^5)}{1-0.5}=13.5625$
  \item $\sum_{n=1}^{10}(40+2n)=400+2(55)=510$
  \item $\tfrac{8(9)(17)}6=204$
  \item $\sum n^2-\sum n=\tfrac{6(7)(13)}6-\tfrac{6(7)}2=91-21=70$
  \item $N^2=225\Rightarrow N=15$
  \item Both give $5+7+9+11+13=45$ --- shifting the index start from 1 to 0 while adjusting the formula's constant leaves every actual term unchanged
  \item $3\sum n^2-2\sum n=3\left(\tfrac{12(13)(25)}6\right)-2\left(\tfrac{12(13)}2\right)=3(650)-2(78)=1950-156=1794$
  \item $15(500)-15\sum_{n=1}^{15}n=7500-15(120)=7500-1800=5700$ units
\end{enumerate}
\end{multicols}
