\begin{ideabox}
Venn diagrams and the addition rule solve "at least one" and unknown-overlap problems.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item $P(A\cup B)=P(A)+P(B)-P(A\cap B)$.
  \item Independent: $P(A\cap B)=P(A)P(B)$. Mutually exclusive: $P(A\cap B)=0$.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Addition rule}
Of 40 students, 24 study French, 18 study Spanish, 9 study both. Find $P(\text{at least one})$.\soln
$P(F\cup S)=\tfrac{24}{40}+\tfrac{18}{40}-\tfrac{9}{40}=\tfrac{33}{40}$.\\[3pt]\textbf{Conclusion:} $\tfrac{33}{40}$.
\end{example}

\begin{example}{ex1}{Example 2 --- Unknown overlap}
Of 60 people, 38 like tea, 30 like coffee, 8 like neither. Find how many like both.\soln
At least one $=52$. $52=38+30-\text{both}\Rightarrow\text{both}=16$.\\[3pt]\textbf{Conclusion:} $16$.
\end{example}

\begin{example}{ex2}{Example 3 --- Independence check}
$P(A)=0.5$, $P(B)=0.4$, $P(A\cap B)=0.2$. Are $A,B$ independent?\soln
$P(A)P(B)=0.2$, matches $P(A\cap B)$.\\[3pt]\textbf{Conclusion:} yes, independent.
\end{example}

\begin{example}{ex3}{Example 4 --- Mutually exclusive check}
$P(C)=0.35$, $P(D)=0.4$, $P(C\cup D)=0.75$. Are $C,D$ mutually exclusive?\soln
$P(C\cap D)=0.35+0.4-0.75=0$.\\[3pt]\textbf{Conclusion:} yes, $P(C\cap D)=0$.
\end{example}

\begin{example}{ex4}{Example 5 --- Finding a region only in one set}
Of 70 people, 45 like hiking, 30 like biking, 12 like both. Find $P(\text{hiking only})$.\soln
Hiking only $=45-12=33$.\\[3pt]\textbf{Conclusion:} $P(\text{hiking only})=\tfrac{33}{70}$.
\end{example}

\begin{example}{ex5}{Example 6 --- Neither region from a Venn diagram}
Of 90 people, 50 like jazz, 40 like classical, 20 like both. Find $P(\text{neither})$.\soln
At least one $=50+40-20=70$; neither $=90-70=20$.\\[3pt]\textbf{Conclusion:} $P(\text{neither})=\tfrac{20}{90}=\tfrac29$.
\end{example}

\begin{example}{ex6}{Example 7 --- Three-fact independence verification}
A die is rolled and a coin is flipped. Let $A$="die shows 4" and $B$="coin shows heads". Verify $A,B$ are independent.\soln
$P(A)=\tfrac16$, $P(B)=\tfrac12$, $P(A\cap B)=\tfrac1{12}$ (one specific outcome out of 12 equally likely).\\[3pt]\textbf{Conclusion:} $P(A)P(B)=\tfrac16\cdot\tfrac12=\tfrac1{12}=P(A\cap B)$ --- confirmed independent (as expected, since a die and coin don't influence each other).
\end{example}

\begin{example}{ex7}{Example 8 --- Solving for a probability given independence}
$A,B$ are independent, $P(A)=0.6$, $P(A\cup B)=0.88$. Find $P(B)$.\soln
$P(A\cup B)=P(A)+P(B)-P(A)P(B)\Rightarrow0.88=0.6+P(B)-0.6P(B)$.\\[3pt]
$0.88=0.6+0.4P(B)\Rightarrow0.28=0.4P(B)$.\\[3pt]\textbf{Conclusion:} $P(B)=0.7$.
\end{example}

\begin{example}{ex8}{Example 9 --- Application --- overlapping subscriptions}
Of 500 households, 320 subscribe to streaming service X, 210 to service Y, and 80 subscribe to neither. Find how many subscribe to both, and $P(\text{exactly one})$.\soln
At least one $=500-80=420$; both $=320+210-420=110$.\\[3pt]\textbf{Conclusion:} both $=110$; exactly one $=420-110=310\Rightarrow P=\tfrac{310}{500}=0.62$.
\end{example}
