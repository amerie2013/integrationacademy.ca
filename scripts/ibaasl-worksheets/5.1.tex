\begin{ideabox}
Limits handle indeterminate forms; the derivative is defined as a limit of secant slopes.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item If direct substitution gives $\tfrac00$, factor or rationalize first.
  \item $f'(x)=\displaystyle\lim_{h\to0}\dfrac{f(x+h)-f(x)}{h}$.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Factoring an indeterminate form}
Evaluate $\displaystyle\lim_{x\to5}\dfrac{x^2-25}{x-5}$.\soln
$\dfrac{(x-5)(x+5)}{x-5}=x+5\to10$.\\[3pt]\textbf{Conclusion:} $10$.
\end{example}

\begin{example}{ex1}{Example 2 --- Rationalizing}
Evaluate $\displaystyle\lim_{x\to0}\dfrac{\sqrt{x+9}-3}{x}$.\soln
Multiply by the conjugate: $\dfrac{x}{x(\sqrt{x+9}+3)}=\dfrac{1}{\sqrt{x+9}+3}\to\dfrac16$.\\[3pt]\textbf{Conclusion:} $\dfrac16$.
\end{example}

\begin{example}{ex2}{Example 3 --- First principles}
Use first principles to find the derivative of $f(x)=x^2-3x$.\soln
$f(x+h)-f(x)=2xh+h^2-3h=h(2x+h-3)$; divide by $h$, let $h\to0$.\\[3pt]\textbf{Conclusion:} $f'(x)=2x-3$.
\end{example}

\begin{example}{ex3}{Example 4 --- General quadratic from first principles}
Use first principles to prove $\dfrac{d}{dx}[ax^2]=2ax$.\soln
$f(x+h)-f(x)=a(x+h)^2-ax^2=a(2xh+h^2)=h(2ax+ah)$; divide by $h$, let $h\to0$.\\[3pt]\textbf{Conclusion:} $f'(x)=2ax$.
\end{example}

\begin{example}{ex4}{Example 5 --- Factoring a higher-degree indeterminate form}
Evaluate $\displaystyle\lim_{x\to3}\dfrac{x^3-27}{x-3}$ using the difference of cubes.\soln
$x^3-27=(x-3)(x^2+3x+9)$, so the limit simplifies to $x^2+3x+9\to9+9+9$.\\[3pt]\textbf{Conclusion:} $27$.
\end{example}

\begin{example}{ex5}{Example 6 --- A limit requiring two factoring steps}
Evaluate $\displaystyle\lim_{x\to2}\dfrac{x^2-4}{x^2-5x+6}$.\soln
Factor both: $\dfrac{(x-2)(x+2)}{(x-2)(x-3)}=\dfrac{x+2}{x-3}\to\dfrac{4}{-1}$.\\[3pt]\textbf{Conclusion:} $-4$.
\end{example}

\begin{example}{ex6}{Example 7 --- First principles for a cubic}
Use first principles to find the derivative of $f(x)=x^3$.\soln
$f(x+h)-f(x)=(x+h)^3-x^3=3x^2h+3xh^2+h^3=h(3x^2+3xh+h^2)$; divide by $h$, let $h\to0$.\\[3pt]\textbf{Conclusion:} $f'(x)=3x^2$.
\end{example}

\begin{example}{ex7}{Example 8 --- Rationalizing with a linear numerator shift}
Evaluate $\displaystyle\lim_{x\to4}\dfrac{\sqrt{x}-2}{x-4}$.\soln
Multiply by the conjugate: $\dfrac{x-4}{(x-4)(\sqrt x+2)}=\dfrac{1}{\sqrt x+2}\to\dfrac1{4}$.\\[3pt]\textbf{Conclusion:} $\dfrac14$.
\end{example}

\begin{example}{ex8}{Example 9 --- First principles at a specific point}
Use first principles to find $f'(3)$ for $f(x)=x^2+1$ (evaluate the limit at the specific point $x=3$).\soln
$\dfrac{f(3+h)-f(3)}h=\dfrac{(3+h)^2+1-10}h=\dfrac{6h+h^2}h=6+h\to6$.\\[3pt]\textbf{Conclusion:} $f'(3)=6$.
\end{example}
