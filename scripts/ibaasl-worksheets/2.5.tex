{\LARGE\bfseries\color{titleblue}Worksheet 2.5 --- The Quadratic Formula and Discriminant}\\[2pt]
{\small Mathematics: Analysis \& Approaches SL (IB) \textperiodcentered{} Unit 2: Functions}

\begin{ideabox}
The discriminant predicts how many real solutions a quadratic has before you finish solving it.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item $x=\dfrac{-b\pm\sqrt{b^2-4ac}}{2a}$.
  \item $\Delta=b^2-4ac$: $\Delta>0$ two real roots, $\Delta=0$ one repeated root, $\Delta<0$ no real roots.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Solve with the formula}
Solve $3x^2-x-4=0$.\soln
$a=3,b=-1,c=-4$: $\Delta=1+48=49$. $x=\dfrac{1\pm7}{6}$.\\[3pt]\textbf{Conclusion:} $x=\dfrac43$ or $x=-1$.
\end{example}

\begin{example}{ex1}{Example 2 --- Nature of the roots}
Determine the nature of the roots of $2x^2+3x+5=0$.\soln
$\Delta=9-40=-31<0$.\\[3pt]\textbf{Conclusion:} no real roots.
\end{example}

\begin{example}{ex2}{Example 3 --- Repeated root condition}
Find $k$ so that $x^2-8x+k=0$ has a repeated root.\soln
$\Delta=0$: $64-4k=0\Rightarrow k=16$.\\[3pt]\textbf{Conclusion:} $k=16$.
\end{example}

\begin{example}{ex3}{Example 4 --- Application}
A ball's height is $h(t)=-5t^2+18t+1$. Determine whether it ever reaches 20 m.\soln
Set $-5t^2+18t+1=20\Rightarrow-5t^2+18t-19=0$. $\Delta=324-380=-56<0$.\\[3pt]\textbf{Conclusion:} no real solution, so it never reaches 20 m.
\end{example}

\begin{example}{ex4}{Example 5 --- Solving using the discriminant to predict, then confirm}
Show that $x^2-6x+9=0$ has a repeated root, and find it.\soln
$\Delta=36-36=0$, confirming a repeated root. $x=\dfrac{6\pm0}{2}=3$.\\[3pt]\textbf{Conclusion:} repeated root at $x=3$.
\end{example}

\begin{example}{ex5}{Example 6 --- Two-parameter repeated-root condition}
Find both values of $m$ so that $x^2+mx+25=0$ has a repeated root.\soln
$\Delta=0$: $m^2-100=0\Rightarrow m=\pm10$.\\[3pt]\textbf{Conclusion:} $m=10$ or $m=-10$.
\end{example}

\begin{example}{ex6}{Example 7 --- Discriminant with a parameter inside a real scenario}
A rectangle's area is $x^2+kx+9=25$ for side length $x$. Find the value(s) of $k$ for which exactly one value of $x$ solves this.\soln
Rearranged: $x^2+kx-16=0$. Exactly one solution requires $\Delta=0$: $k^2+64=0$ --- impossible for real $k$ (since $k^2\geq0$, $k^2+64>0$ always).\\[3pt]\textbf{Conclusion:} no real value of $k$ gives exactly one solution --- this equation always has two distinct real roots (one positive, one negative, since the constant term is negative). ✓
\end{example}

\begin{example}{ex7}{Example 8 --- Comparing two quadratics via discriminants}
Determine which of $x^2-4x+5=0$ and $x^2-4x+3=0$ has real roots.\soln
First: $\Delta=16-20=-4<0$ (no real roots). Second: $\Delta=16-12=4>0$ (two real roots).\\[3pt]\textbf{Conclusion:} only the second equation has real roots.
\end{example}

\begin{example}{ex8}{Example 9 --- Application --- break-even analysis}
A company's profit is $P(x)=-2x^2+40x-150$ (in \$1000s), for $x$ hundred units sold. Determine, using the discriminant, whether the company can ever break even ($P=0$).\soln
$\Delta=1600-1200=400>0$.\\[3pt]\textbf{Conclusion:} yes --- two real break-even points exist (solve $-2x^2+40x-150=0$ to find $x=5$ or $x=15$).
\end{example}
