\begin{ideabox}
Tree diagrams handle sequences of events, including without-replacement draws and reversed conditionals.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item $P(A\mid B)=\dfrac{P(A\cap B)}{P(B)}$.
  \item Multiply along a branch; add across matching outcomes.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Basic conditional probability}
$P(A)=0.6$, $P(A\cap B)=0.18$. Find $P(B\mid A)$.\soln
$P(B\mid A)=\dfrac{0.18}{0.6}$.\\[3pt]\textbf{Conclusion:} $0.3$.
\end{example}

\begin{example}{ex1}{Example 2 --- Without replacement}
A bag has 6 red, 4 blue balls. Two drawn without replacement; find $P(\text{both red})$.\soln
$\dfrac6{10}\times\dfrac59=\dfrac{30}{90}$.\\[3pt]\textbf{Conclusion:} $\dfrac13$.
\end{example}

\begin{example}{ex2}{Example 3 --- At least one}
Using the same bag, find $P(\text{at least one blue})$.\soln
$1-P(\text{both red})=1-\tfrac13$.\\[3pt]\textbf{Conclusion:} $\dfrac23$.
\end{example}

\begin{example}{ex3}{Example 4 --- Reverse conditional (tree reversal)}
Factory A makes 65\% of items with a 2\% defect rate; Factory B makes 35\% with a 6\% defect rate. Given an item is defective, find $P(\text{from B})$.\soln
$P(A\cap d)=0.65(0.02)=0.013$; $P(B\cap d)=0.35(0.06)=0.021$; total $=0.034$.\\[3pt]\textbf{Conclusion:} $P(B\mid d)=\dfrac{0.021}{0.034}\approx0.618$.
\end{example}

\begin{example}{ex4}{Example 5 --- Three-stage without-replacement sequence}
A bag has 5 red, 3 blue balls. Three drawn without replacement; find $P(\text{all three red})$.\soln
$\dfrac58\times\dfrac47\times\dfrac36=\dfrac{60}{336}$.\\[3pt]\textbf{Conclusion:} $\dfrac{60}{336}=\dfrac{5}{28}$.
\end{example}

\begin{example}{ex5}{Example 6 --- Conditional probability from a two-way table}
Of 100 students, 60 play sports; of those, 40 also do music. Of the 40 who don't play sports, 10 do music. Find $P(\text{music}\mid\text{sports})$.\soln
$P(\text{music}\mid\text{sports})=\dfrac{40}{60}$.\\[3pt]\textbf{Conclusion:} $\dfrac23$.
\end{example}

\begin{example}{ex6}{Example 7 --- Comparing conditional probabilities both ways}
Using the same data (100 students; 60 sports, 40 of them music; 40 non-sports, 10 of them music), find $P(\text{sports}\mid\text{music})$ and compare to $P(\text{music}\mid\text{sports})$.\soln
Total music $=40+10=50$. $P(\text{sports}\mid\text{music})=\dfrac{40}{50}=0.8$.\\[3pt]\textbf{Conclusion:} $P(\text{sports}\mid\text{music})=0.8\neq P(\text{music}\mid\text{sports})=\tfrac23\approx0.667$ --- confirming order matters.
\end{example}

\begin{example}{ex7}{Example 8 --- Independence test using a tree diagram's branches}
A spinner has $P(\text{red})=0.4$ on each of two independent spins. Find $P(\text{exactly one red})$ using a tree diagram.\soln
Two paths: (red, not red) and (not red, red), each with probability $0.4\times0.6=0.24$.\\[3pt]\textbf{Conclusion:} $P(\text{exactly one red})=0.24+0.24=0.48$.
\end{example}

\begin{example}{ex8}{Example 9 --- Application --- medical test reliability (reverse conditional)}
A disease affects 2\% of a population. A test correctly detects it 95\% of the time (true positive) and gives a false positive 3\% of the time. Given a positive result, find $P(\text{has the disease})$.\soln
$P(\text{disease}\cap+)=0.02(0.95)=0.019$. $P(\text{no disease}\cap+)=0.98(0.03)=0.0294$. Total $P(+)=0.0484$.\\[3pt]\textbf{Conclusion:} $P(\text{disease}\mid+)=\dfrac{0.019}{0.0484}\approx0.393$ --- surprisingly low, since the disease is rare.
\end{example}

\begin{keybox}
\begin{multicols}{2}
\small
\begin{enumerate}[leftmargin=*,itemsep=3pt]
  \item $0.28/0.7=0.4$
  \item $\tfrac5{10}\times\tfrac49=\tfrac{20}{90}=\tfrac29$
  \item $1-\tfrac29=\tfrac79$
  \item $2(0.3\times0.7)=0.42$
  \item $P(\text{no faulty})=\tfrac9{12}\times\tfrac8{11}=\tfrac{72}{132}=\tfrac6{11}$; at least one $=1-\tfrac6{11}=\tfrac5{11}$
  \item $0.4(0.05)+0.6(0.02)=0.02+0.012=0.032$
  \item $P(X\mid d)=\dfrac{0.02}{0.032}=0.625$
  \item $\tfrac6{12}\times\tfrac5{11}\times\tfrac4{10}=\tfrac{120}{1320}=\tfrac1{11}$
  \item $P(\text{sport}\mid\text{instrument})=\tfrac{30}{70}=\tfrac37\approx0.429$
  \item total sport players $=30+20=50$; $P(\text{instrument}\mid\text{sport})=\tfrac{30}{50}=0.6$; not equal to $\tfrac37\approx0.429$, confirming order matters
  \item $2(0.3\times0.7)=0.42$
  \item $P(d\cap+)=0.01(0.9)=0.009$; $P(\text{no }d\cap+)=0.99(0.04)=0.0396$; total $=0.0486$; $P(d\mid+)=0.009/0.0486\approx0.185$
  \item $P(\text{no faulty})=\tfrac{10}{12}\times\tfrac9{11}=\tfrac{90}{132}=\tfrac{15}{22}$; $P(\text{both faulty})=\tfrac2{12}\times\tfrac1{11}=\tfrac2{132}=\tfrac1{66}$; exactly one $=1-\tfrac{15}{22}-\tfrac1{66}=1-\tfrac{45}{66}-\tfrac1{66}=\tfrac{20}{66}=\tfrac{10}{33}$
\end{enumerate}
\end{multicols}
