\begin{ideabox}
Tree diagrams handle sequences of events, including without-replacement draws and reversed conditionals.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item $P(A\mid B)=\dfrac{P(A\cap B)}{P(B)}$.
  \item Multiply along a branch; add across matching outcomes.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Basic conditional probability}
$P(A)=0.6$, $P(A\cap B)=0.18$. Find $P(B\mid A)$.\soln
$P(B\mid A)=\dfrac{0.18}{0.6}$.\\[3pt]\textbf{Conclusion:} $0.3$.
\end{example}

\begin{example}{ex1}{Example 2 --- Without replacement}
A bag has 6 red, 4 blue balls. Two drawn without replacement; find $P(\text{both red})$.\soln
$\dfrac6{10}\times\dfrac59=\dfrac{30}{90}$.\\[3pt]\textbf{Conclusion:} $\dfrac13$.
\end{example}

\begin{example}{ex2}{Example 3 --- At least one}
Using the same bag, find $P(\text{at least one blue})$.\soln
$1-P(\text{both red})=1-\tfrac13$.\\[3pt]\textbf{Conclusion:} $\dfrac23$.
\end{example}

\begin{example}{ex3}{Example 4 --- Reverse conditional (tree reversal)}
Factory A makes 65\% of items with a 2\% defect rate; Factory B makes 35\% with a 6\% defect rate. Given an item is defective, find $P(\text{from B})$.\soln
$P(A\cap d)=0.65(0.02)=0.013$; $P(B\cap d)=0.35(0.06)=0.021$; total $=0.034$.\\[3pt]\textbf{Conclusion:} $P(B\mid d)=\dfrac{0.021}{0.034}\approx0.618$.
\end{example}

  \item $0.28/0.7=0.4$
  \item $\tfrac5{10}\times\tfrac49=\tfrac{20}{90}=\tfrac29$
  \item $1-\tfrac29=\tfrac79$
  \item $2(0.3\times0.7)=0.42$
  \item $P(\text{no faulty})=\tfrac9{12}\times\tfrac8{11}=\tfrac{72}{132}=\tfrac6{11}$; at least one $=1-\tfrac6{11}=\tfrac5{11}$
  \item $0.4(0.05)+0.6(0.02)=0.02+0.012=0.032$
  \item $P(X\mid d)=\dfrac{0.02}{0.032}=0.625$
\end{enumerate}
