{\LARGE\bfseries\color{titleblue}Worksheet 2.1 --- Function Basics: Domain, Range and Notation}\\[2pt]
{\small Mathematics: Analysis \& Approaches SL (IB) \textperiodcentered{} Unit 2: Functions}
\end{center}

\begin{ideabox}
A function assigns exactly one output to every input; its domain and range describe what inputs and outputs are possible.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item Domain restrictions: avoid division by zero and square roots of negatives.
  \item Vertical line test: a graph is a function iff no vertical line crosses it more than once.
  \item Range: think about the smallest and largest outputs the function can actually reach.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Evaluate a function}
Given $f(x)=3x^2-4x+1$, find $f(-2)$.\soln
$3(4)-4(-2)+1=12+8+1$.\\[3pt]\textbf{Conclusion:} $f(-2)=21$.
\end{example}

\begin{example}{ex1}{Example 2 --- Domain with a root}
Find the domain of $f(x)=\sqrt{2x-8}$.\soln
$2x-8\geq0\Rightarrow x\geq4$.\\[3pt]\textbf{Conclusion:} domain is $x\geq4$.
\end{example}

\begin{example}{ex2}{Example 3 --- Domain with a fraction}
Find the domain of $f(x)=\dfrac{5}{x^2-9}$.\soln
$x^2-9\neq0\Rightarrow x\neq\pm3$.\\[3pt]\textbf{Conclusion:} all real $x$ except $3$ and $-3$.
\end{example}

\begin{example}{ex3}{Example 4 --- Range from a transformed root}
Find the range of $f(x)=-\sqrt{x-2}+5$.\soln
Domain is $x\geq2$; at $x=2$, $f=5$ (largest value, since $-\sqrt{\phantom x}$ falls as $x$ grows).\\[3pt]\textbf{Conclusion:} range is $f(x)\leq5$.
\end{example}

\begin{example}{ex4}{Example 5 --- Combined domain restrictions}
Find the domain of $f(x)=\dfrac{\sqrt{x+2}}{x-3}$.\soln
Need $x+2\geq0\Rightarrow x\geq-2$, and also $x-3\neq0\Rightarrow x\neq3$.\\[3pt]\textbf{Conclusion:} domain is $x\geq-2$, $x\neq3$.
\end{example}

\begin{example}{ex5}{Example 6 --- Range on a restricted domain}
Find the range of $f(x)=x^2-4x+3$ for $0\leq x\leq5$.\soln
\textbf{Step 1:} Vertex at $x=2$ (inside the domain): $f(2)=4-8+3=-1$ (minimum).\\[3pt]
\textbf{Step 2:} Check endpoints: $f(0)=3$, $f(5)=25-20+3=8$.\\[3pt]\textbf{Conclusion:} range is $-1\leq f(x)\leq8$.
\end{example}

\begin{example}{ex6}{Example 7 --- Application --- domain from a real constraint}
A rectangle's area is $A(x)=x(20-x)$, where $x$ is one side and $20-x$ is the other (from a fixed perimeter). State the domain that makes physical sense.\soln
Both sides must be positive: $x>0$ and $20-x>0$.\\[3pt]\textbf{Conclusion:} domain is $0<x<20$.
\end{example}

\begin{example}{ex7}{Example 8 --- Testing a set of ordered pairs}
Determine whether $\{(1,2),(2,3),(1,5)\}$ represents a function.\soln
The input $x=1$ appears twice, with two different outputs ($2$ and $5$).\\[3pt]\textbf{Conclusion:} not a function --- one input must give exactly one output.
\end{example}

\begin{example}{ex8}{Example 9 --- Range of a semicircle function}
Find the range of $f(x)=-\sqrt{16-x^2}$ (domain $-4\leq x\leq4$).\soln
This is the lower half of a circle of radius 4; $f$ ranges from $-4$ (at $x=0$) up to $0$ (at $x=\pm4$).\\[3pt]\textbf{Conclusion:} range is $-4\leq f(x)\leq0$.
\end{example}
