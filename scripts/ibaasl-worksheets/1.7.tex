\begin{ideabox}
A proof must hold for every case a claim covers; a single counterexample is enough to disprove a claim.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item \textbf{Direct proof}: start from known facts/definitions and reason forward to the conclusion.
  \item \textbf{Proof by exhaustion}: check every case, when there are only finitely many.
  \item \textbf{Disproof by counterexample}: one case where the claim fails is enough.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Direct proof}
Prove that the sum of two consecutive even integers is always divisible by 4... or is it? Check first, then prove or disprove.\soln
\textbf{Step 1 (test):} $2+4=6$ (not divisible by 4) --- the claim is false as stated.\\[3pt]
\textbf{Step 2:} Let the two consecutive even integers be $2n$ and $2n+2$. Their sum is $4n+2=2(2n+1)$, which is divisible by 2 but not generally by 4 (since $2n+1$ is odd).\\[3pt]\textbf{Conclusion:} the original claim is false; the sum is always divisible by 2, but not always by 4.
\end{example}

\begin{example}{ex1}{Example 2 --- Disproof by counterexample}
Disprove: "For every integer $n$, $n^2+n+1$ is odd."\soln
Test $n=1$: $1+1+1=3$ (odd). Test $n=2$: $4+2+1=7$ (odd). Test general parity: if $n$ is even, $n^2$ even, $n$ even, so $n^2+n+1$ is even+even+odd=odd. If $n$ is odd, $n^2$ odd, $n$ odd, so odd+odd+odd=odd.\\[3pt]\textbf{Conclusion:} the claim actually holds for every integer --- checking both parities (a proof by exhaustion over the two cases even and odd) proves it rather than disproving it.
\end{example}

\begin{example}{ex2}{Example 3 --- Proof by exhaustion}
Prove that $n^2-n$ is even for every integer $n$ from $1$ to $4$, then generalize.\soln
\textbf{Step 1 (exhaustion for $n=1..4$):} $n=1:0$; $n=2:2$; $n=3:6$; $n=4:12$ --- all even.\\[3pt]
\textbf{Step 2 (general case):} $n^2-n=n(n-1)$, the product of two consecutive integers. One of any two consecutive integers is always even, so their product is always even.\\[3pt]\textbf{Conclusion:} $n^2-n$ is even for every integer $n$.
\end{example}

\begin{example}{ex3}{Example 4 --- Direct proof --- algebraic identity}
Prove that the difference of squares of two consecutive odd integers is always a multiple of 8.\soln
\textbf{Step 1:} Let the odd integers be $2n+1$ and $2n+3$.\\[3pt]
\textbf{Step 2:} $(2n+3)^2-(2n+1)^2=\big[(2n+3)-(2n+1)\big]\big[(2n+3)+(2n+1)\big]=(2)(4n+4)=8n+8=8(n+1)$.\\[3pt]\textbf{Conclusion:} the difference is $8(n+1)$, a multiple of 8 for every integer $n$.
\end{example}

\begin{keybox}
\begin{multicols}{2}
\small
\begin{enumerate}[leftmargin=*,itemsep=3pt]
  \item $n+(n+1)+(n+2)=3n+3=3(n+1)$, a multiple of 3
  \item $n=10$: $100+10+11=121=11^2$, not prime --- disproved
  \item $(2k)^2=4k^2$, divisible by 4
  \item $10=2\cdot5$, $11$ prime, $12=2^2\cdot3$, $13$ prime, $14=2\cdot7$ --- every case checks out
  \item $x=0.5$: $0.25<0.5$ --- disproved
  \item $(2m+1)(2n+1)=4mn+2m+2n+1=2(2mn+m+n)+1$, odd
  \item Among $n-1,n,n+1$ (three consecutive integers) one is divisible by 3 and at least one is even, so their product is divisible by both 2 and 3, hence by 6
  \item False: $a=2,b=-2$ gives $a^2=b^2=4$ but $a\neq b$ --- disproved by counterexample
\end{enumerate}
\end{multicols}
\end{keybox}
