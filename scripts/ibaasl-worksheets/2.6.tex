\begin{ideabox}
Rational functions built from $\dfrac{a}{x-h}+k$ have asymptotes you can read straight from the equation.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item $y=\dfrac1x$ has asymptotes $x=0$ and $y=0$.
  \item $y=\dfrac{a}{x-h}+k$: vertical asymptote $x=h$, horizontal asymptote $y=k$.
\end{itemize}
\end{ideabox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Reciprocal transformation}
State the asymptotes of $y=\dfrac{4}{x-3}+2$.\soln
Vertical: $x=3$. Horizontal: $y=2$.\\[3pt]\textbf{Conclusion:} $x=3$, $y=2$.
\end{example}

\begin{example}{ex1}{Example 2 --- Reciprocal transformation with reflection}
Sketch $y=\dfrac{-2}{x+1}-3$ as a transformation of $y=\dfrac1x$.\soln
Reflected in $x$-axis and stretched by 2; shifted left 1 and down 3.\\[3pt]\textbf{Conclusion:} asymptotes $x=-1$, $y=-3$.
\end{example}

\begin{example}{ex2}{Example 3 --- Domain of a general rational function}
Find the domain and vertical asymptote of $y=\dfrac{2x-1}{x+5}$.\soln
Denominator zero at $x=-5$.\\[3pt]\textbf{Conclusion:} domain all real $x$ except $-5$; vertical asymptote $x=-5$.
\end{example}

\begin{example}{ex3}{Example 4 --- Application}
A club's average cost per member is $C(n)=\dfrac{600}{n}+15$ dollars, for $n$ members. Interpret both asymptotes in context.\soln
Vertical asymptote $n=0$: the model breaks down with zero members. Horizontal asymptote $y=15$: as membership grows, the average cost per member settles near \$15 (the per-person variable cost).\\[3pt]\textbf{Conclusion:} \$15 is the cost floor once fixed costs are spread thin.
\end{example}

\begin{keybox}
\begin{multicols}{2}
\small
\begin{enumerate}[leftmargin=*,itemsep=3pt]
  \item $x=-7$, $y=0$
  \item $x=0$, $y=-4$
  \item $x=2$, $y=6$
  \item all real $x$ except $-9$
  \item $4=\dfrac{a}{1}+1\Rightarrow a=3$
  \item asymptotes $x=-2$, $y=-1$
  \item As volume approaches 0, pressure would grow without bound --- physically, a gas cannot be compressed to zero volume
  \item $5=\dfrac{a}{2}+2\Rightarrow a=6$; intercept: $0=\dfrac{6}{x+3}+2\Rightarrow x+3=-3\Rightarrow x=-6$
\end{enumerate}
\end{multicols}
\end{keybox}
