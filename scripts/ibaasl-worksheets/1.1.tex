\begin{ideabox}
An arithmetic sequence has a constant common difference $d$ between consecutive terms.
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item General term: $u_n=u_1+(n-1)d$.
  \item Sum of the first $n$ terms: $S_n=\dfrac n2\big(2u_1+(n-1)d\big)=\dfrac n2(u_1+u_n)$.
  \item $d>0$ increasing, $d<0$ decreasing, $d=0$ constant.
\end{itemize}
\end{ideabox}

\begin{learnbox}
\lh{Two equivalent sum formulas}$S_n=\dfrac n2(2u_1+(n-1)d)$ uses only $u_1$ and $d$; $S_n=\dfrac n2(u_1+u_n)$ is faster once you already know the last term $u_n$.
\lh{Finding $u_1,d$ from two terms}Two given terms give two equations in $u_1$ and $d$ --- subtract them to eliminate $u_1$ and solve for $d$ first.
\end{learnbox}

\sech{Worked Examples}

\begin{example}{ex0}{Example 1 --- Find a term}
Find $u_{20}$ for the sequence $8,13,18,\dots$\soln
$u_1=8$, $d=5$. $u_{20}=8+(19)(5)=8+95$.\\[3pt]
\textbf{Conclusion:} $u_{20}=103$.
\end{example}

\begin{example}{ex1}{Example 2 --- Solve for the first term and common difference}
An arithmetic sequence has $u_6=23$ and $u_{11}=48$. Find $u_1$ and $d$.\soln
\textbf{Step 1:} Subtract: $u_{11}-u_6=5d=48-23=25\Rightarrow d=5$.\\[3pt]
\textbf{Step 2:} $u_6=u_1+5d=23\Rightarrow u_1+25=23\Rightarrow u_1=-2$.\\[3pt]
\textbf{Conclusion:} $u_1=-2$, $d=5$.
\end{example}

\begin{example}{ex2}{Example 3 --- Sum of a series}
Find $S_{25}$ for $7,11,15,\dots$\soln
$u_1=7$, $d=4$, $n=25$. $S_{25}=\dfrac{25}{2}\big(2(7)+24(4)\big)=\dfrac{25}{2}(14+96)$.\\[3pt]
\textbf{Conclusion:} $S_{25}=\dfrac{25}{2}(110)=1375$.
\end{example}

\begin{example}{ex3}{Example 4 --- Application --- stacking}
Logs are stacked with 22 in the bottom row, each row above having 2 fewer, down to 4 in the top row. Find the total number of logs.\soln
\textbf{Step 1:} $u_1=22$, $d=-2$, last term $u_n=4$.\\[3pt]
\textbf{Step 2:} Find $n$: $4=22+(n-1)(-2)\Rightarrow -18=-2(n-1)\Rightarrow n=10$.\\[3pt]
\textbf{Step 3:} $S_{10}=\dfrac{10}{2}(22+4)=5(26)$.\\[3pt]
\textbf{Conclusion:} $130$ logs.
\end{example}

\begin{example}{ex4}{Example 5 --- Sum from first and last term only}
An arithmetic sequence has $u_1=5$, $d=6$, and last term $u_n=95$. Find $S_n$.\soln
\textbf{Step 1:} Find $n$: $95=5+6(n-1)\Rightarrow 90=6(n-1)\Rightarrow n=16$.\\[3pt]
\textbf{Step 2:} $S_{16}=\dfrac{16}{2}(5+95)=8(100)$.\\[3pt]
\textbf{Conclusion:} $S_{16}=800$.
\end{example}

\begin{example}{ex5}{Example 6 --- Application --- rising salary}
A worker's salary starts at \$38{,}000 and rises by \$1400 each year. Find the total earned over the first 9 years.\soln
$u_1=38000$, $d=1400$, $n=9$. $S_9=\dfrac92\big(2(38000)+8(1400)\big)=\dfrac92(76000+11200)$.\\[3pt]
\textbf{Conclusion:} $S_9=\dfrac92(87200)=\$392{,}400$.
\end{example}

\begin{example}{ex6}{Example 7 --- Solve for $n$ from a sum equation}
An arithmetic sequence has $u_1=2$, $d=6$. Find $n$ such that $S_n=184$.\soln
\textbf{Step 1:} $S_n=\dfrac n2\big(4+6(n-1)\big)=n(3n-1)=3n^2-n$.\\[3pt]
\textbf{Step 2:} $3n^2-n-184=0\Rightarrow(n-8)(3n+23)=0$.\\[3pt]
\textbf{Conclusion:} $n=8$ (rejecting the negative root). ✓
\end{example}

\begin{example}{ex7}{Example 8 --- Two sequences meeting}
Sequence A: $u_1=10$, $d=3$. Sequence B: $u_1=50$, $d=-2$. Find the term number where both sequences give the same value.\soln
\textbf{Step 1:} Set general terms equal: $10+3(n-1)=50-2(n-1)$.\\[3pt]
\textbf{Step 2:} $3n+7=52-2n\Rightarrow5n=45\Rightarrow n=9$.\\[3pt]
\textbf{Conclusion:} at $n=9$, both sequences equal $34$. ✓
\end{example}

\begin{example}{ex8}{Example 9 --- Application --- accumulated temperature drop}
A region's daily low starts at $20°$C on day 1 and drops by $1.5°$C each day. Find the sum of all daily lows over the first 14 days.\soln
$u_1=20$, $d=-1.5$, $n=14$. $S_{14}=\dfrac{14}2\big(2(20)+13(-1.5)\big)=7(40-19.5)$.\\[3pt]
\textbf{Conclusion:} $S_{14}=7(20.5)=143.5°$C (sum of daily lows). ✓
\end{example}

\begin{keybox}
\begin{multicols}{2}
\small
\begin{enumerate}[leftmargin=*,itemsep=3pt]
  \item $u_{15}=62$
  \item $u_{30}=-95$
  \item $d=4$, $u_1=6$
  \item $S_{18}=1107$
  \item $S_{40}=-3700$
  \item $12$ rows: $S_{12}=\tfrac{12}{2}(2(18)+11(3))=6(69)=414$ seats
  \item $5+(n-1)4=161\Rightarrow n=40$ terms
  \item $n$: $98=3+5(n-1)\Rightarrow n=20$; $S_{20}=\tfrac{20}2(3+98)=10(101)=1010$
  \item $S_{20}=\tfrac{20}2(60+19(4))=10(136)=1360$ seats
  \item $S_n=\tfrac n2(6+4(n-1))>200\Rightarrow 2n^2+4n-200>0\Rightarrow n^2+2n-100>0$; testing, $n=10$ gives $S_{10}=210>200$ (and $n=9$ gives $171<200$), so $n=10$
  \item $S_{10}=\tfrac{10}{2}(8+9d)=205\Rightarrow 40+45d=205\Rightarrow d=\tfrac{165}{45}=\tfrac{11}{3}$
  \item $S_n=n(4n+3)-n... $ precisely $4n^2+3n=612\Rightarrow(n-12)(4n+51)=0\Rightarrow n=12$
  \item $100-7(n-1)=4+9(n-1)\Rightarrow107-7n=9n-5\Rightarrow112=16n\Rightarrow n=7$; value $=100-7(6)=58$
\end{enumerate}
\end{multicols}
\end{keybox}
